#pragma once
#include <cstdint>
#include <utility>
#include "../tool/helper.hpp"
#include "../tool/magic_number.hpp"
namespace protocol::IoT {
using tool::Bytes;
struct AudioPacket {
    int64_t timestampMs = 0;
    uint32_t durationMs = 0;
    Bytes payload;
    uint32_t codec = 0;
    uint32_t sampleRate = 0;
    uint32_t channels = 0;
};
namespace client {
inline Bytes encodeAudioPayload(const Bytes& payload, int64_t timestampMs,
                                uint32_t durationMs) {
    Bytes b;
    tool::putU32(b, BUIS_MAGIC(AudioPacket));
    tool::putI64(b, timestampMs);
    tool::putU32(b, durationMs);
    tool::putBytes(b, payload);
    return b;
}
inline Bytes encodeAudio(const AudioPacket& packet) {
    Bytes b;
    tool::putU32(b, BUIS_MAGIC(AudioPacket));
    tool::putI64(b, packet.timestampMs);
    tool::putU32(b, packet.durationMs);
    tool::putBytes(b, packet.payload);
    tool::putU32(b, packet.codec);
    tool::putU32(b, packet.sampleRate);
    tool::putU32(b, packet.channels);
    return b;
}
inline Bytes encodeAudio(const Bytes& payload, int64_t timestampMs, uint32_t durationMs = 20) {
    return encodeAudio(AudioPacket{timestampMs, durationMs, payload, 0, 0, 0});
}
}
namespace server {
inline bool isAudio(const Bytes& b) {
    return tool::hasMagic(b, BUIS_MAGIC(AudioPacket));
}
inline bool decodeAudio(const Bytes& b, AudioPacket& out) {
    tool::Reader r{b};
    AudioPacket tmp;
    if (!tool::open(b, BUIS_MAGIC(AudioPacket), r) ||
        !r.i64(tmp.timestampMs) ||
        tmp.timestampMs < 0 ||
        !r.u32(tmp.durationMs) ||
        !r.bytes(tmp.payload)) {
        return false;
    }
    r.u32(tmp.codec);
    r.u32(tmp.sampleRate);
    r.u32(tmp.channels);
    if (!r.done()) {
        return false;
    }
    out = std::move(tmp);
    return true;
}
}
}

enum class Status_DeviceQuery : uint8_t {
    SUCCESS          = 0,
    DEVICE_NOT_FOUND = 1,
    INVALID_PARAM    = 2,
    SERVER_ERROR     = 3,
    AUTH_REQUIRED    = 4,
    AUTH_FAILED      = 5
};
inline bool isValidStatus(uint8_t raw) {
    return raw <= static_cast<uint8_t>(Status_DeviceQuery::AUTH_FAILED);
}
struct BuisDeviceQueryRequest {
    std::string deviceId;
    bool valid() const noexcept { return !deviceId.empty(); }
};
struct BuisDeviceQueryResponse {
    Status_DeviceQuery status = Status_DeviceQuery::SERVER_ERROR;
    Bytes nonce;
    std::string message;
};
struct BuisDeviceAuthRequest {
    std::string deviceId;
    Bytes hmac;
    bool valid() const noexcept {
        return !deviceId.empty() && hmac.size() == 32;
    }
};
namespace client {

    return tool::hasMagic(b, BUIS_MAGIC(BuisDeviceQueryResponse));
}

    tool::putU32(b, BUIS_MAGIC(BuisDeviceQueryRequest));
    tool::putStr(b, req.deviceId);
    return b;
}

    return encodeDeviceQuery(BuisDeviceQueryRequest{deviceId});
}
inline Bytes encodeDeviceAuth(const BuisDeviceAuthRequest& req) {
    Bytes b;
    tool::putU32(b, BUIS_MAGIC(BuisDeviceAuthRequest));
    tool::putStr(b, req.deviceId);
    tool::putBytes(b, req.hmac);
    return b;
}
inline bool decodeDeviceQueryResponse(const Bytes& b, BuisDeviceQueryResponse& out) {
    tool::Reader r{b};

        !r.u8(rawStatus) ||

    out = std::move(tmp);
    return true;
}

#pragma once
#include <cstdint>
#include <string>
#include <utility>
#include "../tool/helper.hpp"
#include "../tool/magic_number.hpp"
namespace protocol::IoT {
using tool::Bytes;
enum class RecordCommand : uint8_t {
    STOP    = 0,
    START   = 1,
    MAX_CMD = START
};
inline const char* toString(RecordCommand cmd) {
    switch (cmd) {
        case RecordCommand::STOP:  return "STOP";
        case RecordCommand::START: return "START";
        default:                   return "UNKNOWN";
    }
}
struct RecordControlInfo {
    RecordCommand command = RecordCommand::STOP;
    int64_t timestampMs = 0;
    uint32_t requestId = 0;
    bool valid() const noexcept {
        return timestampMs >= 0 &&
               static_cast<uint8_t>(command) <= static_cast<uint8_t>(RecordCommand::MAX_CMD);
    }
};
struct RecordAckInfo {
    RecordCommand command = RecordCommand::STOP;
    int64_t timestampMs = 0;
    uint32_t requestId = 0;
    bool ok = true;
    std::string errorMessage;
    bool valid() const noexcept {
        return timestampMs >= 0 &&
               static_cast<uint8_t>(command) <= static_cast<uint8_t>(RecordCommand::MAX_CMD);
    }
};
namespace client {
inline bool isRecordControl(const Bytes& b) {
    return tool::hasMagic(b, BUIS_MAGIC(RecordControlInfo));
}
inline bool decodeRecordControl(const Bytes& b, RecordControlInfo& out) {
    tool::Reader r{b};
    uint8_t rawCmd = 0;
    RecordControlInfo tmp;
    if (!tool::open(b, BUIS_MAGIC(RecordControlInfo), r) ||
        !r.u8(rawCmd) ||
        rawCmd > static_cast<uint8_t>(RecordCommand::MAX_CMD) ||
        !r.i64(tmp.timestampMs) ||
        tmp.timestampMs < 0) {
        return false;
    }
    r.u32(tmp.requestId);
    if (!r.done()) {
        return false;
    }
    tmp.command = static_cast<RecordCommand>(rawCmd);
    out = std::move(tmp);
    return true;
}
inline Bytes encodeRecordAck(const RecordAckInfo& info) {
    Bytes b;
    tool::putU32(b, BUIS_MAGIC(RecordAckInfo));
    b.push_back(static_cast<uint8_t>(info.command));
    tool::putI64(b, info.timestampMs);
    b.push_back(info.ok ? 1 : 0);
    tool::putStr(b, info.errorMessage);
    tool::putU32(b, info.requestId);
    return b;
}
inline Bytes encodeRecordAck(RecordCommand cmd,
                             int64_t timestampMs = 0,
                             uint32_t requestId = 0,
                             bool ok = true,
                             std::string errorMessage = "") {
    return encodeRecordAck(RecordAckInfo{cmd, timestampMs, requestId, ok, std::move(errorMessage)});
}
}
namespace server {
inline Bytes encodeRecordControl(const RecordControlInfo& info) {
    Bytes b;
    tool::putU32(b, BUIS_MAGIC(RecordControlInfo));
    b.push_back(static_cast<uint8_t>(info.command));
    tool::putI64(b, info.timestampMs);
    if (info.requestId != 0) {
        tool::putU32(b, info.requestId);
    }
    return b;
}
inline Bytes encodeRecordControl(RecordCommand cmd, int64_t timestampMs = 0,
                                 uint32_t requestId = 0) {
    return encodeRecordControl(RecordControlInfo{cmd, timestampMs, requestId});
}
inline bool isRecordAck(const Bytes& b) {
    return tool::hasMagic(b, BUIS_MAGIC(RecordAckInfo));
}
inline bool decodeRecordAck(const Bytes& b, RecordAckInfo& out) {
    tool::Reader r{b};
    uint8_t rawCmd = 0;
    uint8_t rawOk = 0;
    RecordAckInfo tmp;
    if (!tool::open(b, BUIS_MAGIC(RecordAckInfo), r) ||
        !r.u8(rawCmd) ||
        rawCmd > static_cast<uint8_t>(RecordCommand::MAX_CMD) ||
        !r.i64(tmp.timestampMs) ||
        tmp.timestampMs < 0 ||
        !r.u8(rawOk) ||
        !r.str(tmp.errorMessage)) {
        return false;
    }
    r.u32(tmp.requestId);
    if (!r.done()) {
        return false;
    }
    tmp.command = static_cast<RecordCommand>(rawCmd);
    tmp.ok = (rawOk != 0);
    out = std::move(tmp);
    return true;
}
}
}

inline void encodeFileEntries(Bytes& b, const std::vector<FileInfoEntry>& files) {
    tool::putU32(b, static_cast<std::uint32_t>(files.size()));
    for (const auto& f : files) {
        tool::putStr(b, f.deviceId);
        tool::putStr(b, f.fileId);
        tool::putStr(b, f.fileName);
        tool::putI64(b, f.createdTimeMs);
        tool::putI64(b, f.durationMs);
        tool::putI64(b, f.sizeBytes);
        b.push_back(f.isOpen ? 1 : 0);
        tool::putStr(b, f.description);
        tool::putU32(b, f.codec);
        tool::putU32(b, f.sampleRate);
        tool::putU32(b, f.channels);
    }
}
constexpr std::size_t kMinEntryBytes = 41;
inline bool decodeEntry(tool::Reader& r, FileInfoEntry& entry, bool withCodec) {
    std::uint8_t openFlag = 0;
    if (!r.str(entry.deviceId) ||
        !r.str(entry.fileId) ||
        !r.str(entry.fileName) ||
        !r.i64(entry.createdTimeMs) ||
        !r.i64(entry.durationMs) ||
        !r.i64(entry.sizeBytes) ||
        !r.u8(openFlag) ||
        !r.str(entry.description)) {
        return false;
    }
    if (withCodec) {
        if (!r.u32(entry.codec) || !r.u32(entry.sampleRate) || !r.u32(entry.channels)) {
            return false;
        }
    }
    entry.isOpen = (openFlag != 0);
    return true;
}
inline bool decodeFileEntriesLayout(tool::Reader& r, std::size_t totalBytes,
                                    std::uint32_t vectorSize,
                                    std::vector<FileInfoEntry>& out,
                                    bool withCodec) {
    if (static_cast<std::size_t>(vectorSize) > (totalBytes - r.pos) / kMinEntryBytes) {
        return false;
    }
    std::vector<FileInfoEntry> entries;
    entries.reserve(vectorSize);
    for (std::uint32_t i = 0; i < vectorSize; ++i) {
        FileInfoEntry entry;
        if (!decodeEntry(r, entry, withCodec)) {
            return false;
        }
        entries.push_back(std::move(entry));
    }
    out = std::move(entries);
    return true;
}
inline bool decodeFileEntries(tool::Reader& r,
                              std::size_t totalBytes,
                              std::vector<FileInfoEntry>& out) {
    std::uint32_t vectorSize = 0;
    if (!r.u32(vectorSize)) {
        return false;
    }
    const std::size_t bodyStart = r.pos;
    if (decodeFileEntriesLayout(r, totalBytes, vectorSize, out, true) &&
        r.pos == totalBytes) {
        return true;
    }
    r.pos = bodyStart;
    out.clear();
    return decodeFileEntriesLayout(r, totalBytes, vectorSize, out, false) &&
           r.pos == totalBytes;
}

    tool::Reader r{b};

        return false;

    std::uint8_t moreFlag = 0;
    std::uint32_t count = 0;
    if (!r.str(tmp.fileId) ||
        !r.str(tmp.message) ||
        !r.u8(moreFlag) ||
        !r.i64(tmp.nextTimestampMs) ||
        !r.u32(count)) {

 #pragma once
#include <cstddef>
#include <cstdint>
#include <cstdio>

#include <string>

namespace protocol::tool {
using Bytes = std::vector<uint8_t>;
inline void putU32(Bytes& b, uint32_t v) {
    for (int s = 24; s >= 0; s -= 8) b.push_back(uint8_t(v >> s));
}
inline void putStr(Bytes& b, const std::string& s) {
    putU32(b, uint32_t(s.size()));
    b.insert(b.end(), s.begin(), s.end());
}
inline void putBytes(Bytes& b, const Bytes& data) {
    putU32(b, uint32_t(data.size()));
    b.insert(b.end(), data.begin(), data.end());
}
inline void putU64(Bytes& b, uint64_t v) {
    for (int s = 56; s >= 0; s -= 8) b.push_back(uint8_t(v >> s));
}
inline void putI64(Bytes& b, int64_t v) {
    putU64(b, static_cast<uint64_t>(v));
}
struct Reader {
    const Bytes& b;
    size_t pos = 0;
    bool u8(uint8_t& v) {
        if (pos + 1 > b.size()) return false;
        v = b[pos++];
        return true;
    }
    bool u32(uint32_t& v) {
        if (b.size() - pos < 4) return false;
        v = (uint32_t(b[pos]) << 24) | (uint32_t(b[pos + 1]) << 16) |
            (uint32_t(b[pos + 2]) << 8) | uint32_t(b[pos + 3]);
        pos += 4;
        return true;
    }
    bool u64(uint64_t& v) {
        if (b.size() - pos < 8) return false;
        v = 0;
        for (int i = 0; i < 8; ++i) {
            v = (v << 8) | uint64_t(b[pos++]);
        }
        return true;
    }
       bool i64(int64_t& v) {
        uint64_t raw = 0;
        if (!u64(raw)) return false;
        v = static_cast<int64_t>(raw);
        return true;
    }
    bool str(std::string& s) {
        uint32_t len;
        if (!u32(len) || len > b.size() - pos) return false;
        s.assign(b.begin() + pos, b.begin() + pos + len);
        pos += len;
        return true;
    }
        bool bytes(Bytes& out) {
        uint32_t len;
        if (!u32(len) || len > b.size() - pos) return false;
        out.assign(b.begin() + pos, b.begin() + pos + len);
        pos += len;
        return true;
    }
    bool done() const { return pos == b.size(); }
};
inline bool hasMagic(const Bytes& b, uint32_t magic) {
    if (b.size() < 4) return false;
    Reader r{b};
    uint32_t m;
    return r.u32(m) && m == magic;
}
inline bool open(const Bytes& b, uint32_t magic, Reader& r) {
    uint32_t m;
    return r.u32(m) && m == magic;
}

constexpr uint32_t magic(std::string_view name) {
    uint32_t h = 2166136261u;
    for (char c : name) {
        h ^= static_cast<uint8_t>(c);
        h *= 16777619u;
    }
    return h;
}
class MagicRegistry {
public:
    static MagicRegistry& instance() {
        static MagicRegistry r;
        return r;
    }
    void add(uint32_t value, const char* name) {
        std::lock_guard<std::mutex> l(m_);
        auto [it, inserted] = names_.emplace(value, name);
        if (!inserted) {
            std::fprintf(stderr,
                "[protocol] magic number collision 0x%08X: \"%s\" and \"%s\"\n",
                static_cast<unsigned int>(value), it->second.c_str(), name);
            std::abort();
        }
    }
    std::string name_of(uint32_t value) {
        std::lock_guard<std::mutex> l(m_);
        auto it = names_.find(value);
        return it == names_.end() ? "unknown" : it->second;
    }
private:
    std::mutex m_;
    std::unordered_map<uint32_t, std::string> names_;
};
struct MagicRegistrar {
    MagicRegistrar(uint32_t value, const char* name) {
        MagicRegistry::instance().add(value, name);
    }
};
template <std::size_t N>
struct FixedString {
    char data[N]{};
    constexpr FixedString(const char (&s)[N]) {
        for (std::size_t i = 0; i < N; ++i) data[i] = s[i];
    }
    constexpr std::string_view view() const { return {data, N - 1}; }
};
template <FixedString Name>
struct Registered {
    static constexpr uint32_t value = magic(Name.view());
    static uint32_t id() {
        (void)&reg;
        return value;
    }
private:
    [[maybe_unused]] static inline const MagicRegistrar reg{value, Name.data};
};

# architecture

Ports & Adapters (hexagonal) with four folders. Dependencies point **inward**:
the application core knows nothing about ESP-IDF, FreeRTOS, KCP, WiFi or the board.

```
main/
  interface.hpp        ports + AudioConfig + ITransportFactory
  main.cpp             thin entry: hardware::boot(config) + start core/app task
  app_context.*        composition root (plain class) owning adapters + services
  core/                application + domain (pure C++)
    app/               AppLoop, AudioService, NetworkService, media_controller,
                       session_factory
    session/           SessionManager (handshake, liveness, reconnect)
    audio/             AudioPipeline
  general/             portable libraries (no core/ or esp_hardware/ deps)
    kcp/               ikcp + KcpSession + DatagramIo
    crypto/            PSA AES-256-GCM + ECDH P-256 (backend injected)
    opus/              Opus wrapper
    audio/             frame math
    io/                path join, stdio file store, random-access file
    fs/                directory lister
    input/             button debounce / long-press state machine
    net/               POSIX UDP socket helpers (getaddrinfo/sendto/recvfrom)
    util/              MAC / device-id formatting
  esp_hardware/        the ONLY folder allowed to touch ESP-IDF
    hardware.hpp/.cpp  boot(const core::AudioConfig&) + board bring-up
    adapters/          port implementations (incl. media/UI inputs)
    drivers/           bsp, i2s, buttons, led, speech, io expander
    wifi/              WiFi manager / provisioning
    net/               UDP DatagramIo adapter (POSIX socket code is in general/net)
    storage/           LittleFS mount + OTA backend
```

`main/CMakeLists.txt` adds `main/`, `main/general` and `main/esp_hardware` as
include roots, so includes use stable prefixes (`interface.hpp`, `core/...`,
`general/...`, `esp_hardware/...`) regardless of where a file lives.

## Ports (`main/interface.hpp`)

| Port | Responsibility | Today | Tomorrow |
|---|---|---|---|
| `ILink` | start the path + report usable link | `EspWifiLink` | 4G / Ethernet |
| `ITransport` | move bytes to/from server; own conv id | `KcpUdpTransport` | PPP / TCP |
| `ICrypto` | ECDH + AES-GCM | `EspCrypto` | other backend |
| `IClock` | monotonic ms + unix ms | `esp_timer` / `gettimeofday` | any OS |
| `IRandom` | session id | `esp_random` | any OS |
| `ILogger` | logging (`log` / `logf`) | `ESP_LOGx` | any OS |
| `IDeviceInfo` | MAC / chip / device id | `EspDeviceInfo` | any board |
| `IStorage` | small blob key-value | `EspNvsStorage` | any KV store |
| `IFileStore` | mount + file persistence | `EspFileStore` | any FS |
| `IAudioSource` | open/close/read one raw PCM frame | `EspI2sAudioSource` | any mic |
| `IAudioEncoder` | encode raw PCM frame bytes | `OpusEncoder` | LC3 / other |
| `ITaskRunner` | create/stop/delay/exit tasks | `EspTaskRunner` (FreeRTOS) | `std::thread` / RTOS |
| `IFirmwareUpdater` | OTA init / apply | `EspFirmwareUpdater` | SD / serial |
| `IMediaPlayer` | play/pause/stop/volume | `EspMediaPlayer` | any player |
| `IIndicator` | RGB / display status | `EspIndicator` | any UI |
| `IButtonInput` | physical buttons | `EspButtonInput` | any input |
| `ISpeechInput` | voice commands | `EspSpeechInput` | any ASR |
| `IPlaylist` | track listing | `EspSdPlaylist` | any FS |
| `ITransportFactory` | create the session transport | `AppContext` | any transport |
| `INetwork` | start/send/receive/epoch of the session | `core::app::NetworkService` | any session |
| `IAudioService` | start/stop/isRunning capture | `core::app::AudioService` | any capture |

`AppContext` owns two service objects built from the ports: `NetworkService`
(transport + `SessionManager` + worker task) and `AudioService`
(capture/encode/send). Each takes only the ports it uses; there is no
`IAppServices` service locator and no global "current" network/audio state.
`AudioConfig` carries the sample rate / channels / bit depth / frame duration.

Audio frames cross the ports as raw `Bytes`: `IAudioSource::readFrame(Bytes&)`
returns exactly the bytes the I2S/codec configuration produces, and
`IAudioEncoder::encode(const Bytes&, Bytes&)` consumes them. The slot layout
(2 x 16-bit vs 4 x 16-bit vs 32-bit words) is a board/encoder contract, not
something `AudioPipeline` re-interprets as `int16` samples.

## General logic rescued from `esp_hardware`

Deleting `main/esp_hardware/` now only removes board glue; these behaviours are
compiled from `core/` and `general/`:

| Logic | Lives in | Hardware counterpart |
|---|---|---|
| KCP session, tuning, reassembly, liveness | `general/kcp/KcpSession` | `net/EspUdpDatagramIo` (over `general/net` POSIX sockets) |
| playlist, volume clamp, play/pause, voice feedback | `core/app/MediaController` | `hardware.cpp` + drivers |
| button debounce / short / long press | `general/input/ButtonStateMachine` | `drivers/button` |
| directory scan + extension filter | `general/fs/DirectoryLister` | `drivers/bsp` |
| path join, stdio read/write | `general/io/Path`, `general/io/StdioFileStore` | `adapters/storage/EspFileStore` |
| random-access file ops (offset/append/shift/truncate) | `general/io/RandomAccessFile` | `storage/file_manager` |
| MAC / device-id formatting | `general/util/format.hpp` | `adapters/device/EspDeviceInfo` |
| session wiring / conv id | `core/app/session_factory` + `NetworkService` | `AppContext::create` |
| audio frame sizing | `general/audio/frame_math.hpp` | `drivers/audio_capture` |

## Layer rules (enforced)

`tools/check_layers.sh` (grep + include gate):

* `core/` never mentions ESP-IDF, FreeRTOS, NVS, WiFi or PSA.
* `core/` never includes `app_context.hpp`, `esp_hardware/` or `adapters/`.
* `general/` never mentions ESP-IDF, FreeRTOS, NVS or WiFi. PSA/Opus/KCP are
  allowed: they are portable libraries, not ESP-IDF headers.
* `general/` never includes `interface.hpp`, `core/` or `esp_hardware/`.
* `esp_hardware/` never includes `core/`; it may use `interface.hpp`, `general/`
  and ESP-IDF.
* `main/` (`app_context`, `main.cpp`) is the composition root and may include
  everything.

`tools/host_check.sh` (host compiler, C++20):

* `core/app` (incl. `prog.cpp` and `audio_service.cpp`), `core/audio`,
  `core/session` compile with `-fsyntax-only` on the host using the proto
  headers (C++20 is required by `components/proto`).
* `general/` pure sources + `general/kcp/ikcp.c` (C99) compile on the host.
* The POSIX UDP `DatagramIo` adapter (`esp_hardware/net`) is host-checked too.
* `general/crypto` and `general/opus` need PSA/libopus headers, so they are
  validated by the IDF build.

`tools/run_host_tests.sh` (host unit tests, no hardware):

* `ButtonStateMachine`: short press, long press fires once while held, and a
  release exactly at the threshold still yields `LongPress`.
* `AudioPipeline`: one encode per source frame, monotonic timestamps (first
  frame is 0), source exhaustion stops the loop.
* `SessionManager` with fake `ITransport` / `ILink` / `IClock` / `IRandom` /
  `ICrypto`: handshake -> established, encrypted `sendReliable`, link-down
  reset, reopen throttle, silent-server reset with a fresh conv id, WaitAck
  timeout, dead-link early reset + epoch increments.

Run the three gates (layers, host compile, host tests) after every change. Run `idf.py reconfigure` after adding a source
directory (the CMake glob has no `CONFIGURE_DEPENDS`).

## Domain core (`main/core`)

* `session/SessionManager` - handshake / liveness / reconnect / encryption:
  ```
  IDLE --link up--> HANDSHAKING(WAIT_ACK) --peer key--> ESTABLISHED
  ESTABLISHED --no RX for 8s--> IDLE (reset transport => fresh KCP sequence numbers)
  any --link down--> IDLE (reset transport)
  ```
  * handshake retries forever with backoff 600ms -> 3s (server may boot late)
  * every reset reopens the transport with a FRESH conversation id (random.u32)
    so a restarted/alive server and the client never desync KCP sequences
  * KCP dead_link (retransmission limit) triggers an early reset instead of
    waiting for the full 8 s server timeout
  * transport.open() is attempted only while the link is up (no getaddrinfo
    stalls when offline)
  * inbound messages are drained in a loop per tick; KCP datagrams drained
    per poll with a 32-datagram bound
  * outbound data queued (max 128) and dropped when not established
    (real-time audio: dropping is better than buffering)
* `audio/AudioPipeline` - capture -> encode -> emit `(compressed, timestampMs)`.

## Application (`main/core/app`)

* `prog.cpp` - `core::app::AppLoop`: device registration + server message
  dispatch. Takes `INetwork&` / `IAudioService&` plus the ports it uses; no
  ESP-IDF/root includes.
  Registration is sent on (re)connect transitions detected via the session
  epoch, then as a 3 s keepalive; the RX queue is drained per loop iteration;
  `IFirmwareUpdater::markAppValid()` runs once on the first successful query.
* `audio_service.cpp` - `core::app::AudioService`: start/stop the capture task
  through `ITaskRunner`; record acks go out through the injected `INetwork`
  and use `IClock::nowUnixMs()`.
* `network_service.cpp` - `core::app::NetworkService`: builds the
  transport/session via `ITransportFactory` + `SessionManager`, then runs the
  worker task through `ITaskRunner`.

## Hardware (`main/esp_hardware`)

* `hardware.cpp` - board bring-up: BSP/I2S, IO expander and SD card.
  `main.cpp` includes it through the `HARDWARE_HEADER` macro.
* `adapters/` - `KcpUdpTransport`, `EspWifiLink`, `EspCrypto`, `EspClock`,
  `EspRandom`, `EspLogger`, `EspTaskRunner`, `EspDeviceInfo`,
  `EspNvsStorage`, `EspFileStore`, `EspFirmwareUpdater`,
  `EspI2sAudioSource`, `OpusEncoder`, `EspMediaPlayer`, `EspIndicator`,
  `EspButtonInput`, `EspSpeechInput`, `EspSdPlaylist`.
* `drivers/`, `wifi/`, `net/`, `storage/` - the ESP-IDF implementation
  details behind those adapters.

## Composition root

`main/app_context.{hpp,cpp}` is a **plain class** owned by `main.cpp`. It
constructs every concrete adapter plus `NetworkService`, `AudioService` and
`AppLoop`, and passes only the ports each one needs. There is no
`app::context()` singleton: `main.cpp` passes `&context.loop` (typed
`AppLoop*`) to `ITaskRunner::start`, so `core/app` never sees the composition
root.

**Swapping hardware** means replacing `main/esp_hardware/` (or pointing
`HARDWARE_HEADER` at another folder). `main/core/` and `main/general/` stay
untouched; `main/main.cpp` only calls `hardware::boot(context.audio_config)`,
starts the media controller and the application task.

## Crypto backend

`general/crypto` holds the AES-256-GCM / ECDH protocol logic on top of PSA
(mbedTLS is a portable library, so that code stays general).
`encryption::CryptoBackend` is the only platform seam:
`esp_hardware/adapters/crypto/EspCrypto.hpp` implements `EspCryptoBackend`
(`psa_crypto_init()` + `psa_generate_random()`), so a different MCU only
replaces that thin wrapper.

                           ITaskRunner& tasks, ILogger& logger, INetwork& network)
    : input_(input),

}
void AudioService::clearTaskHandle() {
    std::lock_guard<std::mutex> lock(mutex_);
    task_.store(nullptr);
    run_state_.store(RunState::Stopped);
}
void AudioService::taskLoop() {
    const uint32_t request_id = request_id_.load();
    logger_.logf(LogLevel::Info, "Audio task created & initializing");
    if (!input_.open()) {
        logger_.logf(LogLevel::Error, "Audio input open failed");
        sendAck(network_, clock_, protocol::IoT::RecordCommand::START, request_id,
                false, "audio input open failed");
        clearTaskHandle();
        tasks_.exitCurrentTask();
        return;
    }
    if (!encoder_.init()) {
        logger_.logf(LogLevel::Error, "Audio encoder init failed");
        input_.close();
        sendAck(network_, clock_, protocol::IoT::RecordCommand::START, request_id,
                false, "opus encoder init failed");
        clearTaskHandle();
        tasks_.exitCurrentTask();
        return;
    }
    sendAck(network_, clock_, protocol::IoT::RecordCommand::START, request_id, true);
    AudioPipeline pipeline(
        input_, encoder_, clock_, logger_,
        [this](const Bytes& compressed, uint32_t timestamp_ms) {
            network_.send(protocol::IoT::client::encodeAudio(
                protocol::IoT::AudioPacket{
                    static_cast<int64_t>(timestamp_ms),
                    encoder_.frameDurationMs(),
                    compressed,
                    1,
                    encoder_.sampleRate(),
                    encoder_.channelCount()}));
        });
    while (running_.load()) {
        if (!pipeline.step()) {
            tasks_.delayMs(kLoopDelayMs);
        }
    }
    input_.close();
    logger_.logf(LogLevel::Info, "Exiting audio loop: releasing drivers");
    clearTaskHandle();
    tasks_.exitCurrentTask();
}
void AudioService::start(uint32_t request_id) {
    if (run_state_.load() == RunState::Stopping) {
        int waited = 0;
        while (task_.load() != nullptr && waited < kStopTimeoutMs) {
            tasks_.delayMs(kLoopDelayMs);
            waited += kLoopDelayMs;
        }
    }
    std::lock_guard<std::mutex> lock(mutex_);
    if (run_state_.load() != RunState::Stopped) {
        logger_.logf(LogLevel::Warn, "Audio already running; acking START");
        sendAck(network_, clock_, protocol::IoT::RecordCommand::START, request_id, true);
        return;
    }
    request_id_.store(request_id);
    running_.store(true);
    run_state_.store(RunState::Running);
    task_.store(tasks_.start("audio_proc_task", kTaskStack, kTaskPriority, kTaskCore,
                             taskEntry, this));
    if (task_.load() == nullptr) {
        logger_.logf(LogLevel::Error, "Failed to create audio task!");
        run_state_.store(RunState::Stopped);
        running_.store(false);
        sendAck(network_, clock_, protocol::IoT::RecordCommand::START, request_id,
                false, "task create failed");
        return;
    }
}
void AudioService::stop(uint32_t request_id, bool ack) {
    if (run_state_.load() == RunState::Stopped) {
        logger_.logf(LogLevel::Warn, "Audio already stopped");
        if (ack) {
            sendAck(network_, clock_, protocol::IoT::RecordCommand::STOP, request_id, true);
        }
        return;
    }
    logger_.logf(LogLevel::Info, "Stopping audio task...");
    run_state_.store(RunState::Stopping);
    running_.store(false);
    int waited = 0;
    while (task_.load() != nullptr && waited < kStopTimeoutMs) {
        tasks_.delayMs(kLoopDelayMs);
        waited += kLoopDelayMs;
    }
    if (task_.load() != nullptr) {
        logger_.logf(LogLevel::Error,
                     "Audio task did not exit within %d ms; leaving it to finish",
                     kStopTimeoutMs);
        return;
    }
    run_state_.store(RunState::Stopped);
    logger_.logf(LogLevel::Info, "Audio task stopped");
    if (ack) {
        sendAck(network_, clock_, protocol::IoT::RecordCommand::STOP, request_id, true);
    }
}
bool AudioService::isRunning() const {
    return run_state_.load() != RunState::Stopped;
}

        if (binding.id == id) {

void MediaController::perform(MediaAction action) {
    switch (action) {
        case MediaAction::VolumeUp:
            changeVolume(+static_cast<int>(config_.volume_step));
            break;
        case MediaAction::VolumeDown:
            changeVolume(-static_cast<int>(config_.volume_step));
            break;
        case MediaAction::TogglePlayPause:
            resume_after_voice_ = false;
            togglePlayPause();
            break;
        case MediaAction::Stop:
            resume_after_voice_ = false;
            player_.stop();
            indicator_.setMode(core::IndicatorMode::Idle);
            break;
        case MediaAction::NextTrack:
            resume_after_voice_ = false;
            if (!tracks_.empty()) {
                index_ = (index_ + 1) % tracks_.size();
                playCurrent();
            }
            break;
        case MediaAction::PreviousTrack:
            resume_after_voice_ = false;
            if (!tracks_.empty()) {
                index_ = (index_ + tracks_.size() - 1) % tracks_.size();
                playCurrent();
            }
            break;
        case MediaAction::None:
            break;
    }
}
void MediaController::togglePlayPause() {
    switch (player_.state()) {
        case core::MediaState::Running:
            player_.pause();
            indicator_.setMode(core::IndicatorMode::Idle);
            break;
        case core::MediaState::Paused:
            player_.resume();
            indicator_.setMode(core::IndicatorMode::Playing);
            break;
        case core::MediaState::Stopped:
        case core::MediaState::None:
            playCurrent();
            break;
        case core::MediaState::Finished:
            playCurrent();
            break;
        case core::MediaState::Error:
            playCurrent();
            break;
    }
}
void MediaController::playCurrent() {
    if (tracks_.empty()) {
        logger_.logf(core::LogLevel::Warn, "Media: no tracks to play");
        return;
    }
    const std::string url = config_.url_prefix + tracks_[index_];
    if (!player_.play(url)) {
        logger_.logf(core::LogLevel::Error, "Media: failed to play %s", url.c_str());
        return;
    }
    indicator_.setMode(core::IndicatorMode::Playing);
    logger_.logf(core::LogLevel::Info, "Media: playing %s", url.c_str());
}
void MediaController::changeVolume(int delta) {
    int next = static_cast<int>(player_.volume()) + delta;
    if (next < 0) {
        next = 0;
    }
    if (next > static_cast<int>(config_.max_volume)) {
        next = config_.max_volume;
    }
    player_.setVolume(static_cast<uint8_t>(next));
}
void MediaController::onSpeech(core::SpeechEvent event, int command) {
    std::lock_guard<std::mutex> lock(mutex_);
    switch (event) {
        case core::SpeechEvent::Awake:
            if (player_.state() == core::MediaState::Running) {
                player_.pause();
                resume_after_voice_ = true;
            }
            indicator_.setMode(core::IndicatorMode::VoiceCommand);
            break;
        case core::SpeechEvent::Command:
            switch (command) {
                case 0: indicator_.setColor(core::IndicatorColor::Red); break;
                case 1: indicator_.setColor(core::IndicatorColor::Blue); break;
                case 2: indicator_.setColor(core::IndicatorColor::Green); break;
                case 3: indicator_.setColor(core::IndicatorColor::White); break;
                default: break;
            }
            break;
        case core::SpeechEvent::CommandTimeout:
            if (resume_after_voice_) {
                player_.resume();
                indicator_.setMode(core::IndicatorMode::Playing);
                resume_after_voice_ = false;
            } else {
                indicator_.setMode(core::IndicatorMode::Idle);
            }
            break;
    }
}

NetworkService::NetworkService(ITransportFactory& factory, ILink& link,
                               IClock& clock, IRandom& random, ICrypto& crypto,
                               ITaskRunner& tasks, ILogger& logger)
    : factory_(factory),
      link_(link),

NetworkService::~NetworkService() {
    running_.store(false);

void NetworkService::workerEntry(void* arg) {
    static_cast<NetworkService*>(arg)->workerLoop();
}
void NetworkService::workerLoop() {
    logger_.logf(LogLevel::Info, "Network worker task started");
    while (running_.load()) {
        if (session_) {
            session_->tick();
        }
        tasks_.delayMs(kWorkerTickMs);
    }
    {
        std::lock_guard<std::mutex> lock(mutex_);
        handle_.store(nullptr);
    }
    tasks_.exitCurrentTask();
}
void NetworkService::start(const char* host, const char* port) {
    std::lock_guard<std::mutex> lock(mutex_);
    if (running_.load()) {
        logger_.logf(LogLevel::Warn, "Network already initialized");
        return;
    }
    transport_ = factory_.create(host, port, nextConversationId(random_));
    if (!transport_) {
        logger_.logf(LogLevel::Error, "Failed to create transport");
        return;
    }
    session_ = makeSession(*transport_, link_, clock_, random_, crypto_, logger_);
    if (!session_) {
        logger_.logf(LogLevel::Error, "Failed to create session");
        transport_.reset();
        return;
    }
    session_->start();
    running_.store(true);
    handle_.store(tasks_.start("net_worker_io", kWorkerStackBytes, kWorkerPriority,
                               kWorkerCore, workerEntry, this));
    if (handle_.load() == nullptr) {
        logger_.logf(LogLevel::Error, "Failed to create network worker task");
        running_.store(false);
        session_->stop();
        session_.reset();
        transport_.reset();
        return;
    }
    logger_.logf(LogLevel::Info, "Network manager initialized");
}
void NetworkService::close() {
    running_.store(false);
    int waited_ms = 0;
    while (handle_.load() != nullptr && waited_ms < kWorkerExitWaitMs) {
        tasks_.delayMs(10);
        waited_ms += 10;
    }
    if (handle_.load() != nullptr) {
        logger_.logf(LogLevel::Error,
                     "Network worker did not exit; skipping session teardown");
        return;
    }
    {
        std::lock_guard<std::mutex> lock(mutex_);
        if (session_) {
            session_->stop();
            session_.reset();
        }
        transport_.reset();
    }
    logger_.logf(LogLevel::Info, "Network connection closed");
}
int NetworkService::send(const Bytes& data) {
    std::lock_guard<std::mutex> lock(mutex_);
    return session_ ? session_->send(data) : -1;
}
std::optional<Bytes> NetworkService::receive() {
    std::lock_guard<std::mutex> lock(mutex_);
    if (!session_) {
        return std::nullopt;
    }
    return session_->poll();
}
bool NetworkService::isEstablished() const {
    std::lock_guard<std::mutex> lock(mutex_);
    return session_ && session_->isEstablished();
}
uint32_t NetworkService::epoch() const {
    std::lock_guard<std::mutex> lock(mutex_);
    return session_ ? session_->epoch() : 0;
}

int hexNibble(char c) {
    if (c >= '0' && c <= '9') return c - '0';
    if (c >= 'a' && c <= 'f') return c - 'a' + 10;
    if (c >= 'A' && c <= 'F') return c - 'A' + 10;
    return -1;
}

const std::string& AppLoop::effectiveDeviceId()
{
    if (!device_id_loaded_) {
        device_id_loaded_ = true;
        Bytes blob;
        if (storage_.get("device_id", blob) && !blob.empty()) {
            while (!blob.empty() &&
                   (blob.back() == 0 || blob.back() == '\n' || blob.back() == '\r')) {
                blob.pop_back();
            }
            device_id_.assign(blob.begin(), blob.end());
        }
        if (device_id_.empty()) {
            device_id_ = device_.deviceId();
        }
    }
    return device_id_;
}
bool AppLoop::ensureSecretKey()
{
    if (secret_loaded_) {
        return secret_key_.has_value() && !secret_key_->empty();
    }
    secret_loaded_ = true;
    Bytes blob;
    if (!storage_.get("device_secret", blob) || blob.empty()) {
        logger_.logf(LogLevel::Warn,
                     "No device secret in NVS (storage/device_secret); "
                     "challenge-response authentication disabled");
        return false;
    }
    Bytes raw;
    if (blob.size() == 32) {
        raw = std::move(blob);
    } else if (blob.size() == 64) {
        raw.resize(32);
        for (std::size_t i = 0; i < raw.size(); ++i) {
            const int hi = hexNibble(static_cast<char>(blob[i * 2]));
            const int lo = hexNibble(static_cast<char>(blob[i * 2 + 1]));
            if (hi < 0 || lo < 0) {
                logger_.logf(LogLevel::Error,
                             "device_secret contains non-hex characters");
                return false;
            }
            raw[i] = static_cast<uint8_t>((hi << 4) | lo);
        }
    } else {
        logger_.logf(LogLevel::Error,
                     "device_secret must be 32 raw bytes or 64 hex chars (got %u)",
                     static_cast<unsigned>(blob.size()));
        return false;
    }
    auto digest = crypto_.sha256(raw);
    if (!digest.has_value() || digest->size() != 32) {
        logger_.logf(LogLevel::Error, "sha256(device_secret) failed");
        return false;
    }
    secret_key_ = std::move(*digest);
    return true;
}
bool AppLoop::sendDeviceAuth(const Bytes& nonce)
{
    if (nonce.size() != 16) {
        logger_.logf(LogLevel::Error,
                     "Device challenge nonce is %u bytes (expected 16)",
                     static_cast<unsigned>(nonce.size()));
        return false;
    }
    if (!ensureSecretKey()) {
        return false;
    }
    const std::string& device_id = effectiveDeviceId();
    Bytes message = nonce;
    message.insert(message.end(), device_id.begin(), device_id.end());
    auto hmac = crypto_.hmacSha256(*secret_key_, message);
    if (!hmac.has_value() || hmac->size() != 32) {
        logger_.logf(LogLevel::Error, "HMAC-SHA256(device secret) failed");
        return false;
    }
    protocol::IoT::BuisDeviceAuthRequest request;
    request.deviceId = device_id;
    request.hmac = std::move(*hmac);
    network_.send(protocol::IoT::client::encodeDeviceAuth(request));
    logger_.logf(LogLevel::Info, "Device challenge response sent");
    return true;
}
void AppLoop::handleIncoming(const Bytes& packet) {
    if (protocol::IoT::client::isRecordControl(packet)) {
        protocol::IoT::RecordControlInfo info;
        if (!protocol::IoT::client::decodeRecordControl(packet, info)) {
            logger_.logf(LogLevel::Warn, "Failed to decode valid RecordControl payload.");
            return;
        }
        if (info.command == protocol::IoT::RecordCommand::START) {
            logger_.logf(LogLevel::Info,
                         "Record START (requestId=%u, ts=%lld)",
                         static_cast<unsigned>(info.requestId),
                         static_cast<long long>(info.timestampMs));
            audio_.start(info.requestId);
        } else if (info.command == protocol::IoT::RecordCommand::STOP) {
            logger_.logf(LogLevel::Info,
                         "Record STOP (requestId=%u)",
                         static_cast<unsigned>(info.requestId));
            audio_.stop(info.requestId);
        }
        return;
    }
    if (protocol::IoT::client::isDeviceQueryResponse(packet)) {
        protocol::IoT::BuisDeviceQueryResponse response;
        if (!protocol::IoT::client::decodeDeviceQueryResponse(packet, response)) {
            logger_.logf(LogLevel::Warn, "Failed to decode BuisDeviceQueryResponse payload.");
            return;
        }
        if (response.status == protocol::IoT::Status_DeviceQuery::SUCCESS) {
            logger_.logf(LogLevel::Info, "Device query acknowledged: %s",
                         response.message.c_str());
            if (!firmware_marked_ && firmware_.markAppValid()) {
                firmware_marked_ = true;
                logger_.logf(LogLevel::Info, "Firmware marked valid");
            }
        } else if (response.status == protocol::IoT::Status_DeviceQuery::AUTH_REQUIRED) {
            logger_.logf(LogLevel::Info,
                         "Device query requires authentication; answering challenge");
            if (!sendDeviceAuth(response.nonce)) {
                logger_.logf(LogLevel::Warn,
                             "Cannot answer device challenge (missing secret or crypto error)");
            }
        } else {
            logger_.logf(LogLevel::Warn, "Device query rejected (status %u): %s",
                         static_cast<unsigned>(response.status), response.message.c_str());
        }
        return;
    }
    logger_.logf(LogLevel::Info, "Received general server message or unhandled packet type.");
}
void AppLoop::run() {
    if (!files_.init()) {
        logger_.logf(LogLevel::Warn, "Filesystem mount failed");
    }
    firmware_.init();
    link_.start();
    network_.start(host_.c_str(), port_.c_str());
    const std::string& device_id = effectiveDeviceId();
    const std::vector<uint8_t> registration =
        protocol::IoT::client::encodeDeviceQuery(device_id);
    logger_.logf(LogLevel::Info, "Device identity: %s", device_id.c_str());
    bool registered = false;
    uint32_t last_epoch = 0;
    uint32_t last_registration_ms = 0;
    while (true) {
        const uint32_t epoch = network_.epoch();
        if (epoch != last_epoch) {
            last_epoch = epoch;
            registered = false;
            if (audio_.isRunning()) {
                audio_.stop(0, false);
                logger_.logf(LogLevel::Info, "Audio stopped after session reset");
            }
        }
        if (network_.isEstablished() &&
            (!registered ||
             (clock_.nowMs() - last_registration_ms) >= kKeepaliveMs)) {
            network_.send(registration);
            registered = true;
            last_registration_ms = clock_.nowMs();
        }
        while (auto msg = network_.receive()) {
            handleIncoming(*msg);
        }
        tasks_.delayMs(kLoopDelayMs);
    }
    tasks_.exitCurrentTask();
}

AudioPipeline::AudioPipeline(IAudioSource& source, IAudioEncoder& encoder, IClock& clock,
                             ILogger& logger, Sender sender)
    : source_(source),
      encoder_(encoder),
      clock_(clock),
      logger_(logger),
      sender_(std::move(sender)) {}
void AudioPipeline::reset() {
    frame_count_ = 0;
}
bool AudioPipeline::step() {
    if (!source_.readFrame(pcm_)) {
        return false;
    }
    Bytes compressed;
    if (!encoder_.encode(pcm_, compressed)) {
        char buf[96];
        std::snprintf(buf, sizeof(buf), "Encode failed (frame %u)",
                      static_cast<unsigned>(frame_count_));
        logger_.log(LogLevel::Error, buf);
        return false;
    }
    const uint32_t timestamp_ms = frame_count_ * encoder_.frameDurationMs();
    if (sender_) {
        sender_(compressed, timestamp_ms);
    }
    ++frame_count_;
    return true;
}

#include "core/session/SessionManager.hpp"
#include <cstdarg>
#include <cstdio>
#include <cstring>
#include <utility>
namespace core {
namespace {
constexpr uint32_t kHandshakeBaseRetryMs = 600;
constexpr uint32_t kHandshakeMaxRetryMs  = 3000;
constexpr uint32_t kServerTimeoutMs      = 8000;
constexpr uint32_t kHandshakeTimeoutMs   = 30000;
constexpr uint32_t kReopenThrottleMs     = 1000;
constexpr size_t   kMaxOutboundQueue     = 128;
}
SessionManager::SessionManager(ITransport& transport, ILink& link, IClock& clock,
                               IRandom& random, ICrypto& crypto, ILogger& logger)
    : transport_(transport),
      link_(link),
      clock_(clock),
      random_(random),
      crypto_(crypto),
      logger_(logger) {}
SessionManager::~SessionManager() { stop(); }
void SessionManager::logf(LogLevel level, const char* fmt, ...) {
    char buf[192];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(buf, sizeof(buf), fmt, ap);
    va_end(ap);
    logger_.log(level, buf);
}
void SessionManager::start() {
    state_.store(State::Idle);
    epoch_.store(0);
    aes_key_.clear();
    handshake_pkt_.clear();
    retry_count_ = 0;
    last_attempt_ms_ = 0;
    waitack_start_ms_ = 0;
    last_open_attempt_ms_ = 0;
    {
        std::lock_guard<std::mutex> lk(tx_mutex_);
        incoming_.clear();
        sending_.clear();
    }
    {
        std::lock_guard<std::mutex> lk(rx_mutex_);
        std::queue<Bytes> empty;
        std::swap(rx_queue_, empty);
    }
    if (link_.isUp() && !transport_.open()) {
        logf(LogLevel::Error, "Transport open failed; will keep retrying");
    }
}
void SessionManager::stop() {
    transport_.close();
    state_.store(State::Idle);
}
int SessionManager::send(const Bytes& app_message) {
    if (app_message.empty()) {
        return 0;
    }
    std::lock_guard<std::mutex> lk(tx_mutex_);
    if (incoming_.size() >= kMaxOutboundQueue) {
        return -1;
    }
    incoming_.push_back(app_message);
    return 0;
}
std::optional<Bytes> SessionManager::poll() {
    std::lock_guard<std::mutex> lk(rx_mutex_);
    if (rx_queue_.empty()) {
        return std::nullopt;
    }
    Bytes out = std::move(rx_queue_.front());
    rx_queue_.pop();
    return out;
}
void SessionManager::tick() {
    const uint32_t now = clock_.nowMs();
    if (!transport_.isOpen() && link_.isUp()) {
        if (last_open_attempt_ms_ == 0 || (now - last_open_attempt_ms_) >= kReopenThrottleMs) {
            last_open_attempt_ms_ = now;
            if (transport_.open()) {
                logf(LogLevel::Info, "Transport opened");
            }
        }
    }
    bool transport_ok = transport_.isOpen();
    if (transport_ok) {
        transport_.tick(now);
        if (state_.load() != State::Idle && transport_.isDead()) {
            resetSession("dead link");
            transport_ok = transport_.isOpen();
        }
        if (transport_ok) {
            handleHandshake(now);
            while (auto msg = transport_.receive(now)) {
                handleIncoming(std::move(*msg), now);
            }
        }
        if (state_.load() == State::Established) {
            const uint32_t last_rx = transport_.lastRxMs();
            if (last_rx != 0 &&
                static_cast<int32_t>(now - last_rx) > static_cast<int32_t>(kServerTimeoutMs)) {
                logf(LogLevel::Warn, "Server silent for %u ms; reconnecting",
                     static_cast<unsigned>(now - last_rx));
                resetSession("server timeout");
            }
        }
    }
    {
        std::lock_guard<std::mutex> lk(tx_mutex_);
        if (!incoming_.empty()) {
            sending_.swap(incoming_);
        }
    }
    if (!sending_.empty()) {
        const bool online = transport_ok && link_.isUp() && state_.load() == State::Established;
        if (online) {
            for (const auto& pkt : sending_) {
                auto encrypted = crypto_.encrypt(pkt, aes_key_);
                if (!encrypted.has_value()) {
                    logf(LogLevel::Error, "Encrypt failed; dropping packet");
                    continue;
                }
                if (transport_.sendReliable(*encrypted) < 0) {
                    logf(LogLevel::Warn, "Transport send saturated; dropping packet");
                }
            }
        } else {
            logf(LogLevel::Warn, "Dropped %u packet(s): session not established",
                 static_cast<unsigned>(sending_.size()));
        }
        sending_.clear();
    }
}
void SessionManager::handleHandshake(uint32_t now_ms) {
    if (!link_.isUp()) {
        if (state_.load() != State::Idle) {
            resetSession("link down");
        }
        return;
    }
    switch (state_.load()) {
    case State::Idle: {
        if (last_attempt_ms_ != 0 && (now_ms - last_attempt_ms_) < kHandshakeBaseRetryMs) {
            return;
        }
        auto pub = crypto_.generatePublicKey();
        if (!pub.has_value() || pub->empty()) {
            logf(LogLevel::Error, "ECDH key generation failed");
            return;
        }
        const uint32_t conv = transport_.conversationId();
        handshake_pkt_.resize(sizeof(uint32_t) + pub->size());
        std::memcpy(handshake_pkt_.data(), &conv, sizeof(uint32_t));
        std::memcpy(handshake_pkt_.data() + sizeof(uint32_t), pub->data(), pub->size());
        if (transport_.sendRaw(handshake_pkt_) < 0) {
            logf(LogLevel::Warn, "Handshake send failed; staying in Idle");
            last_attempt_ms_ = now_ms;
            return;
        }
        last_attempt_ms_ = now_ms;
        retry_count_ = 1;
        state_.store(State::WaitAck);
        waitack_start_ms_ = now_ms;
        logf(LogLevel::Info, "Handshake dispatched (conv=%u)", static_cast<unsigned>(conv));
        break;
    }
    case State::WaitAck: {
        if ((now_ms - waitack_start_ms_) > kHandshakeTimeoutMs) {
            logf(LogLevel::Warn, "Handshake timed out; restarting with a fresh keypair");
            resetSession("handshake timeout");
            return;
        }
        uint32_t interval = retry_count_ * kHandshakeBaseRetryMs;
        if (interval > kHandshakeMaxRetryMs) {
            interval = kHandshakeMaxRetryMs;
        }
        if ((now_ms - last_attempt_ms_) >= interval) {
            retry_count_++;
            transport_.sendRaw(handshake_pkt_);
            last_attempt_ms_ = now_ms;
            logf(LogLevel::Warn, "Handshake ACK timeout; retry #%u",
                 static_cast<unsigned>(retry_count_));
        }
        break;
    }
    case State::Established:
        break;
    }
}
void SessionManager::handleIncoming(Bytes message, uint32_t now_ms) {
    (void)now_ms;
    if (state_.load() == State::WaitAck) {
        auto secret = crypto_.computeSharedSecret(message);
        if (secret.has_value() && !secret->empty()) {
            aes_key_ = std::move(*secret);
            handshake_pkt_.clear();
            retry_count_ = 0;
            state_.store(State::Established);
            logf(LogLevel::Info, "Handshake complete; AES-256-GCM key established");
        } else {
            logf(LogLevel::Error, "Corrupt handshake response");
        }
        return;
    }
    if (state_.load() == State::Established) {
        auto decrypted = crypto_.decrypt(message, aes_key_);
        if (!decrypted.has_value()) {
            logf(LogLevel::Error, "Failed to decrypt incoming packet");
            return;
        }
        std::lock_guard<std::mutex> lk(rx_mutex_);
        rx_queue_.push(std::move(*decrypted));
    }
}
void SessionManager::resetSession(const char* reason) {
    logf(LogLevel::Warn, "Session reset (%s)", reason);
    state_.store(State::Idle);
    aes_key_.clear();
    handshake_pkt_.clear();
    retry_count_ = 0;
    last_attempt_ms_ = 0;
    waitack_start_ms_ = 0;
    if (link_.isUp()) {
        transport_.reset(random_.u32());
    } else {
        transport_.close();
    }
    epoch_.fetch_add(1);
    last_open_attempt_ms_ = clock_.nowMs();
}

    }

class EspCrypto : public core::ICrypto {
public:
    std::optional<core::Bytes> generatePublicKey() override {
        return ecdh_.generatePublicKey(backend_);
    }
    std::optional<core::Bytes> computeSharedSecret(const core::Bytes& peer_public_key) override {
        return ecdh_.computeSharedSecret(peer_public_key, backend_);
    }
    std::optional<core::Bytes> encrypt(const core::Bytes& plaintext,
                                       const core::Bytes& key) override {
        return encryption::aes_gcm_encrypt(plaintext, key, backend_);
    }
    std::optional<core::Bytes> decrypt(const core::Bytes& packet,
                                       const core::Bytes& key) override {
        return encryption::aes_gcm_decrypt(packet, key, backend_);
    }
    std::optional<core::Bytes> sha256(const core::Bytes& data) override {
        if (psa_crypto_init() != PSA_SUCCESS) {
            return std::nullopt;
        }
        core::Bytes out(32);
        std::size_t out_len = 0;
        if (psa_hash_compute(PSA_ALG_SHA_256, data.data(), data.size(),
                             out.data(), out.size(), &out_len) != PSA_SUCCESS ||
            out_len != out.size()) {
            return std::nullopt;
        }
        return out;
    }
    std::optional<core::Bytes> hmacSha256(const core::Bytes& key,
                                          const core::Bytes& data) override {
        if (psa_crypto_init() != PSA_SUCCESS || key.empty()) {
            return std::nullopt;
        }
        psa_key_attributes_t attr = PSA_KEY_ATTRIBUTES_INIT;
        psa_set_key_type(&attr, PSA_KEY_TYPE_HMAC);
        psa_set_key_bits(&attr, key.size() * 8);
        psa_set_key_usage_flags(&attr, PSA_KEY_USAGE_SIGN_MESSAGE);
        psa_set_key_algorithm(&attr, PSA_ALG_HMAC(PSA_ALG_SHA_256));
        psa_key_id_t key_id = 0;
        if (psa_import_key(&attr, key.data(), key.size(), &key_id) != PSA_SUCCESS) {
            psa_reset_key_attributes(&attr);
            return std::nullopt;
        }
        core::Bytes out(32);
        std::size_t out_len = 0;
        const psa_status_t status =
            psa_mac_compute(key_id, PSA_ALG_HMAC(PSA_ALG_SHA_256),
                            data.data(), data.size(),
                            out.data(), out.size(), &out_len);
        psa_destroy_key(key_id);
        psa_reset_key_attributes(&attr);
        if (status != PSA_SUCCESS || out_len != out.size()) {
            return std::nullopt;
        }
        return out;
    }
private:
    EspCryptoBackend       backend_;
    encryption::EcdhHandshake ecdh_;
};

#include "interface.hpp"
#include <atomic>
#include <cstring>
#include "esp_netif_sntp.h"
#include "esp_timer.h"
#include "WiFiManager.hpp"
#include "wifi/shared.hpp"
namespace adapters {
class EspWifiLink : public core::ILink {
public:
    bool start() override {
        wifi::init();
        static bool sntp_started = false;
        if (!sntp_started) {
            esp_sntp_config_t config = ESP_NETIF_SNTP_DEFAULT_CONFIG("pool.ntp.org");
            sntp_started = (esp_netif_sntp_init(&config) == ESP_OK);
        }
        return true;
    }
    bool isUp() const override {
        const uint32_t now = static_cast<uint32_t>(esp_timer_get_time() / 1000ULL);
        const uint32_t last = last_check_ms_.load();
        if (last != 0 && (now - last) < kCacheMs) {
            return last_state_.load();
        }
        wifi_status_t status{};
        const bool up = wifi::isConnected() &&
                        wifi::getStatus(status) &&
                        status.ip[0] != '\0' &&
                        std::strcmp(status.ip, "0.0.0.0") != 0;
        last_state_.store(up);
        last_check_ms_.store(now);
        return up;
    }
private:
    static constexpr uint32_t kCacheMs = 500;
    mutable std::atomic<uint32_t> last_check_ms_{0};
    mutable std::atomic<bool>     last_state_{false};
};
}

public:
    TaskId start(const char* name, uint32_t stack_bytes, int priority, int core,
                 void (*fn)(void*), void* arg) override {
        TaskHandle_t handle = nullptr;
        const BaseType_t created =
            (core < 0)
                ? xTaskCreate(fn, name, stack_bytes, arg, priority, &handle)
                : xTaskCreatePinnedToCore(fn, name, stack_bytes, arg, priority, &handle, core);
        if (created != pdPASS) {
            return nullptr;
        }
        return static_cast<TaskId>(handle);
    }
    void stop(TaskId task) override {
        if (task != nullptr) {
            vTaskDelete(static_cast<TaskHandle_t>(task));
        }
    }
    void delayMs(uint32_t ms) override { vTaskDelay(pdMS_TO_TICKS(ms)); }
    void exitCurrentTask() override { vTaskDelete(nullptr); }
};
}

#pragma once
#include "interface.hpp"
#include <string>
#include <utility>
#include "nvs.h"
#include "nvs_flash.h"
namespace adapters {
class EspNvsStorage : public core::IStorage {
public:
    explicit EspNvsStorage(std::string namespace_name)
        : namespace_(std::move(namespace_name)) {
        esp_err_t err = nvs_flash_init();
        if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
            nvs_flash_erase();
            nvs_flash_init();
        }
    }
    bool set(const std::string& key, const core::Bytes& value) override {
        nvs_handle_t h;
        if (nvs_open(namespace_.c_str(), NVS_READWRITE, &h) != ESP_OK) {
            return false;
        }
        bool ok = (nvs_set_blob(h, key.c_str(), value.data(), value.size()) == ESP_OK);
        ok = ok && (nvs_commit(h) == ESP_OK);
        nvs_close(h);
        return ok;
    }
    bool get(const std::string& key, core::Bytes& out) override {
        nvs_handle_t h;
        if (nvs_open(namespace_.c_str(), NVS_READONLY, &h) != ESP_OK) {
            return false;
        }
        size_t len = 0;
        esp_err_t err = nvs_get_blob(h, key.c_str(), nullptr, &len);
        if (err == ESP_ERR_NVS_TYPE_MISMATCH) {
            err = nvs_get_str(h, key.c_str(), nullptr, &len);
            if (err == ESP_OK && len > 0) {
                out.resize(len);
                err = nvs_get_str(h, key.c_str(),
                                  reinterpret_cast<char*>(out.data()), &len);
                if (err == ESP_OK) {
                    out.resize(len - 1);
                }
            }
            nvs_close(h);
            if (err == ESP_OK) {
                return true;
            }
            out.clear();
            return false;
        }
        if (err != ESP_OK) {
            nvs_close(h);
            return false;
        }
        out.resize(len);
        err = nvs_get_blob(h, key.c_str(), out.data(), &len);
        nvs_close(h);
        if (err != ESP_OK) {
            out.clear();
            return false;
        }
        out.resize(len);
        return true;
    }
    bool remove(const std::string& key) override {
        nvs_handle_t h;
        if (nvs_open(namespace_.c_str(), NVS_READWRITE, &h) != ESP_OK) {
            return false;
        }
        bool ok = (nvs_erase_key(h, key.c_str()) == ESP_OK);
        ok = ok && (nvs_commit(h) == ESP_OK);
        nvs_close(h);
        return ok;
    }
    bool exists(const std::string& key) override {
        nvs_handle_t h;
        if (nvs_open(namespace_.c_str(), NVS_READONLY, &h) != ESP_OK) {
            return false;
        }
        size_t len = 0;
        esp_err_t err = nvs_get_blob(h, key.c_str(), nullptr, &len);
        nvs_close(h);
        return err == ESP_OK;
    }
private:
    std::string namespace_;
};
}

#include "adapters/transport/KcpUdpTransport.hpp"
#include "esp_hardware/net/esp_udp_datagram_io.hpp"
namespace adapters {
KcpUdpTransport::KcpUdpTransport(const char* host, const char* port, uint32_t conv_id)
    : host_(host ? host : ""), port_(port ? port : ""), conv_id_(conv_id) {}
KcpUdpTransport::~KcpUdpTransport() { close(); }
bool KcpUdpTransport::open() {
    if (kcp_ && kcp_->isOpen()) {
        return true;
    }
    io_ = std::make_unique<EspUdpDatagramIo>(host_.c_str(), port_.c_str());
    if (!io_->isOpen()) {
        io_.reset();
        return false;
    }
    general::kcp::KcpConfig config;
    config.conversation_id = conv_id_;
    kcp_ = std::make_unique<general::kcp::KcpSession>(*io_, config);
    if (!kcp_->open()) {
        kcp_.reset();
        io_.reset();
        return false;
    }
    return true;
}
void KcpUdpTransport::close() {
    if (kcp_) {
        kcp_->close();
        kcp_.reset();
    }
    io_.reset();
}
bool KcpUdpTransport::isOpen() const {
    return kcp_ && kcp_->isOpen();
}
void KcpUdpTransport::reset(uint32_t new_conv_id) {
    conv_id_ = new_conv_id;
    close();
    open();
}
void KcpUdpTransport::tick(uint32_t now_ms) {
    if (kcp_) {
        kcp_->update(now_ms);
    }
}
int KcpUdpTransport::sendReliable(const core::Bytes& data) {
    if (!kcp_ || data.empty()) {
        return -1;
    }
    return kcp_->send(data.data(), data.size());
}
int KcpUdpTransport::sendRaw(const core::Bytes& data) {
    if (!io_ || data.empty()) {
        return -1;
    }
    return io_->send(data.data(), data.size());
}
std::optional<core::Bytes> KcpUdpTransport::receive(uint32_t now_ms) {
    if (!kcp_) {
        return std::nullopt;
    }
    auto message = kcp_->poll(now_ms);
    if (message.empty()) {
        return std::nullopt;
    }
    return message;
}
uint32_t KcpUdpTransport::lastRxMs() const {
    return kcp_ ? kcp_->lastRxMs() : 0;
}
}

#include "audio_capture_driver.hpp"
#include "esp_heap_caps.h"
#include "esp_log.h"
#include "bsp_board.h"
#include "driver/i2s_std.h"
#include "portmacro.h"
#include <cstring>
#include "general/audio/frame_math.hpp"
static const char* TAG = "AudioCaptureDriver";
namespace Hardware {
AudioCaptureDriver::AudioCaptureDriver(const core::AudioConfig& config)
    : config_(config) {
    channel_count_ = config_.channel_count;
    chunk_size_ = general::audio::samplesPerFrame(config_.sample_rate,
                                                  config_.target_frame_duration_ms);
    const std::size_t buffer_bytes = general::audio::bytesPerFrame(
        config_.sample_rate, config_.target_frame_duration_ms,
        config_.channel_count, config_.bits_per_chan);
    raw_dma_buffer_ = static_cast<int16_t*>(
        heap_caps_malloc(buffer_bytes, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT));
    if (raw_dma_buffer_ == nullptr) {
        ESP_LOGE(TAG, "Failed to allocate %zu bytes in PSRAM for audio buffer", buffer_bytes);
    } else {
        ESP_LOGI(TAG, "Audio Hardware Capture initialized (%zu channels, %zu samples/channel, %lu ms duration)",
                 channel_count_, chunk_size_,
                 static_cast<unsigned long>(config_.target_frame_duration_ms));
    }
}
AudioCaptureDriver::~AudioCaptureDriver() {
    freeBuffer();
}
AudioCaptureDriver::AudioCaptureDriver(AudioCaptureDriver&& other) noexcept
    : config_(other.config_),
      channel_count_(other.channel_count_),
      chunk_size_(other.chunk_size_),
      raw_dma_buffer_(other.raw_dma_buffer_) {
    other.raw_dma_buffer_ = nullptr;
    other.channel_count_ = 0;
    other.chunk_size_ = 0;
}
AudioCaptureDriver& AudioCaptureDriver::operator=(AudioCaptureDriver&& other) noexcept {
    if (this != &other) {
        freeBuffer();
        config_ = other.config_;
        channel_count_ = other.channel_count_;
        chunk_size_ = other.chunk_size_;
        raw_dma_buffer_ = other.raw_dma_buffer_;
        other.raw_dma_buffer_ = nullptr;
        other.channel_count_ = 0;
        other.chunk_size_ = 0;
    }
    return *this;
}
void AudioCaptureDriver::freeBuffer() noexcept {
    if (raw_dma_buffer_ != nullptr) {
        heap_caps_free(raw_dma_buffer_);
        raw_dma_buffer_ = nullptr;
    }
}
esp_err_t AudioCaptureDriver::readSamples() {
    if (raw_dma_buffer_ == nullptr) {
        return ESP_ERR_NO_MEM;
    }
    const std::size_t buffer_bytes = general::audio::bytesPerFrame(
        config_.sample_rate, config_.target_frame_duration_ms,
        config_.channel_count, config_.bits_per_chan);
    size_t bytes_read = 0;
    esp_err_t err = i2s_channel_read(bsp_i2s_get_rx_handle(), raw_dma_buffer_,
                                     buffer_bytes, &bytes_read,
                                     100);
    if (err != ESP_OK || bytes_read == 0) {
        ESP_LOGW(TAG, "Failed to read PCM data directly from I2S");
        return ESP_FAIL;
    }
    if (bytes_read != buffer_bytes) {
        ESP_LOGW(TAG, "Short I2S read: %u / %u bytes",
                 static_cast<unsigned>(bytes_read),
                 static_cast<unsigned>(buffer_bytes));
        return ESP_FAIL;
    }
    return ESP_OK;
}
bool AudioCaptureDriver::copyFrame(std::vector<uint8_t>& out) const {
    if (raw_dma_buffer_ == nullptr) {
        return false;
    }
    const std::size_t frame_bytes = chunk_size_ * channel_count_ *
                                    (config_.bits_per_chan / 8u);
    const auto* bytes = reinterpret_cast<const uint8_t*>(raw_dma_buffer_);
    if (config_.bits_per_chan == config_.pcm_bits_per_chan) {
        out.assign(bytes, bytes + frame_bytes);
        return true;
    }
    if (config_.bits_per_chan == 32 && config_.pcm_bits_per_chan == 16 &&
        channel_count_ == 2) {
        const std::size_t words = frame_bytes / sizeof(uint32_t);
        out.resize(words * sizeof(int16_t));
        auto* dst = reinterpret_cast<int16_t*>(out.data());
        for (std::size_t i = 0; i < words; ++i) {
            uint32_t word = 0;
            std::memcpy(&word, bytes + i * sizeof(uint32_t), sizeof word);
            dst[i] = static_cast<int16_t>(static_cast<uint16_t>(word >> 16));
        }
        return true;
    }
    ESP_LOGE(TAG,
             "Unsupported PCM conversion: i2s=%u bits, pcm=%u bits, channels=%zu",
             static_cast<unsigned>(config_.bits_per_chan),
             static_cast<unsigned>(config_.pcm_bits_per_chan),
             channel_count_);
    return false;
}
}

#include "audio_driver.h"
#include "tca9555_driver.h"
#include "bsp_board.h"
static const char *TAG = "audio play";
static uint8_t Volume = PLAYER_VOLUME;
static esp_asp_handle_t handle = NULL;
static void Audio_PA_EN(void)
{
    Set_EXIO(IO_EXPANDER_PIN_NUM_8,true);
    vTaskDelay(pdMS_TO_TICKS(10));
}
static void Audio_PA_DIS(void)
{
    Set_EXIO(IO_EXPANDER_PIN_NUM_8,false);
    vTaskDelay(pdMS_TO_TICKS(10));
}
static int out_data_callback(uint8_t *data, int data_size, void *ctx)
{
    esp_audio_play((int16_t*)data,data_size,500 / portTICK_PERIOD_MS);
    return 0;
}
static int in_data_callback(uint8_t *data, int data_size, void *ctx)
{
    int ret = fread(data, 1, data_size, ctx);
    ESP_LOGD(TAG, "%s-%d,rd size:%d", __func__, __LINE__, ret);
    return ret;
}
static int mock_event_callback(esp_asp_event_pkt_t *event, void *ctx)
{
    if (event->type == ESP_ASP_EVENT_TYPE_MUSIC_INFO)
    {
        esp_asp_music_info_t info = {0};
        memcpy(&info, event->payload, event->payload_size);
        ESP_LOGI(TAG, "Get info, rate:%d, channels:%d, bits:%d ,bitrate = %d", info.sample_rate, info.channels, info.bits,info.bitrate);
    }
    else if (event->type == ESP_ASP_EVENT_TYPE_STATE)
    {
        esp_asp_state_t st = 0;
        memcpy(&st, event->payload, event->payload_size);
        ESP_LOGI(TAG, "Get State, %d,%s", st, esp_audio_simple_player_state_to_str(st));
    }
    return 0;
}
static void pipeline_init(void)
{
    esp_log_level_set("*", ESP_LOG_INFO);
    esp_asp_cfg_t cfg = {
        .in.cb = NULL,
        .in.user_ctx = NULL,
        .out.cb = out_data_callback,
        .out.user_ctx = NULL,
    };
    esp_gmf_err_t err = esp_audio_simple_player_new(&cfg, &handle);
    if (err != ESP_GMF_ERR_OK || handle == NULL) {
        ESP_LOGE(TAG, "esp_audio_simple_player_new failed: %d", err);
        return;
    }
    err = esp_audio_simple_player_set_event(handle, mock_event_callback, NULL);
    if (err != ESP_GMF_ERR_OK) {
        ESP_LOGE(TAG, "esp_audio_simple_player_set_event failed: %d", err);
    }
}
esp_gmf_err_t Audio_Play_Music(const char* url)
{
    if (handle == NULL) return ESP_GMF_ERR_INVALID_ARG;
    esp_audio_simple_player_stop(handle);
    esp_gmf_err_t err = esp_audio_simple_player_run(handle, url, NULL);
    if (err == ESP_GMF_ERR_OK) {
        Audio_PA_EN();
    } else {
        ESP_LOGE(TAG, "esp_audio_simple_player_run failed: %d", err);
    }
    return err;
}
esp_gmf_err_t Audio_Stop_Play(void)
{
    if (handle == NULL) return ESP_GMF_ERR_INVALID_ARG;
    esp_gmf_err_t err = esp_audio_simple_player_stop(handle);
    Audio_PA_DIS();
    return err;
}
esp_gmf_err_t Audio_Resume_Play(void)
{
    if (handle == NULL) return ESP_GMF_ERR_INVALID_ARG;
    esp_gmf_err_t err = esp_audio_simple_player_resume(handle);
    if (err == ESP_GMF_ERR_OK) {
        Audio_PA_EN();
    }
    return err;
}
esp_gmf_err_t Audio_Pause_Play(void)
{
    if (handle == NULL) return ESP_GMF_ERR_INVALID_ARG;
    esp_gmf_err_t err = esp_audio_simple_player_pause(handle);
    Audio_PA_DIS();
    return err;
}
esp_asp_state_t Audio_Get_Current_State(void)
{
    if (handle == NULL) return ESP_ASP_STATE_NONE;
    esp_asp_state_t state;
    esp_gmf_err_t err = esp_audio_simple_player_get_state(handle, &state);
    if (err != ESP_GMF_ERR_OK) {
        ESP_LOGE("AUDIO", "Get state failed: %d", err);
        return ESP_ASP_STATE_ERROR;
    }
    return state;
}
void Audio_Play_Init(void)
{
    pipeline_init();
}
bool Audio_Player_Is_Ready(void)
{
    return handle != NULL;
}

    }
    else
    {

    }

esp_err_t bsp_codec_adc_init(int sample_rate)
{

    audio_codec_i2s_cfg_t i2s_cfg = {
        .port = I2S_NUM_1,
#if ESP_IDF_VERSION >= ESP_IDF_VERSION_VAL(5, 0, 0)
        .rx_handle = rx_handle,
        .tx_handle = NULL,
#endif
    };
    record_data_if = audio_codec_new_i2s_data(&i2s_cfg);
    audio_codec_i2c_cfg_t i2c_cfg = {.addr = ES7210_CODEC_DEFAULT_ADDR,.bus_handle = i2c_bus_handle};
    record_ctrl_if = audio_codec_new_i2c_ctrl(&i2c_cfg);
    es7210_codec_cfg_t es7210_cfg = {
        .ctrl_if = record_ctrl_if,
        .mic_selected = ES7210_SEL_MIC1 | ES7210_SEL_MIC2 | ES7210_SEL_MIC3 | ES7210_SEL_MIC4,
    };
    record_codec_if = es7210_codec_new(&es7210_cfg);

        .data_if = record_data_if,

    esp_codec_dev_sample_info_t fs = {
        .sample_rate = sample_rate,
        .channel = 2,
        .bits_per_sample = 32,

    esp_codec_dev_set_in_channel_gain(record_dev, ESP_CODEC_DEV_MAKE_CHANNEL_MASK(0), RECORD_VOLUME);
    esp_codec_dev_set_in_channel_gain(record_dev, ESP_CODEC_DEV_MAKE_CHANNEL_MASK(1), RECORD_VOLUME);

static esp_err_t bsp_i2s_init(int i2s_num, uint32_t sample_rate, int channel_format, int bits_per_chan)
{
    esp_err_t ret_val = ESP_OK;
#if ESP_IDF_VERSION >= ESP_IDF_VERSION_VAL(5, 0, 0)
    i2s_slot_mode_t channel_fmt = I2S_SLOT_MODE_STEREO;
    if (channel_format == 1) {
        channel_fmt = I2S_SLOT_MODE_MONO;
    } else if (channel_format == 2) {
        channel_fmt = I2S_SLOT_MODE_STEREO;
    } else {
        ESP_LOGE(TAG, "Unable to configure channel_format %d", channel_format);
        channel_format = 1;
        channel_fmt = I2S_SLOT_MODE_MONO;
    }
    if (bits_per_chan != 16 && bits_per_chan != 32) {
        ESP_LOGE(TAG, "Unable to configure bits_per_chan %d", bits_per_chan);
        bits_per_chan = 32;
    }
    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(i2s_num, I2S_ROLE_MASTER);
    ret_val |= i2s_new_channel(&chan_cfg, &tx_handle, &rx_handle);
    i2s_std_config_t std_cfg = I2S_CONFIG_DEFAULT(sample_rate, channel_fmt, bits_per_chan);
    ret_val |= i2s_channel_init_std_mode(tx_handle, &std_cfg);
    ret_val |= i2s_channel_init_std_mode(rx_handle, &std_cfg);
    ret_val |= i2s_channel_enable(tx_handle);
    ret_val |= i2s_channel_enable(rx_handle);
#else
    i2s_channel_fmt_t channel_fmt = I2S_CHANNEL_FMT_RIGHT_LEFT;
    if (channel_format == 1) {
        channel_fmt = I2S_CHANNEL_FMT_ONLY_LEFT;
    } else if (channel_format == 2) {
        channel_fmt = I2S_CHANNEL_FMT_RIGHT_LEFT;
    } else {
        ESP_LOGE(TAG, "Unable to configure channel_format %d", channel_format);
        channel_format = 1;
        channel_fmt = I2S_CHANNEL_FMT_ONLY_LEFT;
    }
    if (bits_per_chan != 16 && bits_per_chan != 32) {
        ESP_LOGE(TAG, "Unable to configure bits_per_chan %d", bits_per_chan);
        bits_per_chan = 16;
    }
    i2s_config_t i2s_config = I2S_CONFIG_DEFAULT(sample_rate, channel_fmt, bits_per_chan);
    i2s_pin_config_t pin_config = {
        .bck_io_num = GPIO_I2S_SCLK,
        .ws_io_num = GPIO_I2S_LRCK,
        .data_out_num = GPIO_I2S_DOUT,
        .data_in_num = GPIO_I2S_SDIN,
        .mck_io_num = GPIO_I2S_MCLK,
    };
    ret_val |= i2s_driver_install(i2s_num, &i2s_config, 0, NULL);
    ret_val |= i2s_set_pin(i2s_num, &pin_config);
#endif
    return ret_val;
}

esp_err_t esp_audio_play(const int16_t* data, int length, uint32_t ticks_to_wait)
{
    size_t bytes_write = 0;
    esp_err_t ret = ESP_OK;
    if (!play_dev || data == NULL || length <= 0) {
        return ESP_ERR_INVALID_ARG;
    }
    int out_length= length;
    int audio_time = 1;
    if (s_play_sample_rate > 0 && s_play_sample_rate < 16000) {
        audio_time = 16000 / s_play_sample_rate;
    }
    if (s_play_channel_format == 1) {
        audio_time *= 2;
    }
    int *data_out = NULL;
    if (s_bits_per_chan != 32) {
        out_length = length * 2;
        data_out = malloc(out_length);
        if (data_out == NULL) {
            return ESP_ERR_NO_MEM;
        }
        for (int i = 0; i < (int)(length / sizeof(int16_t)); i++) {
            int ret = data[i];
            data_out[i] = ret << 16;
        }
    }
    int *data_out_1 = NULL;
    if (s_play_channel_format != 2 || s_play_sample_rate != 16000) {
        out_length *= audio_time;
        data_out_1 = malloc(out_length);
        if (data_out_1 == NULL) {
            if (data_out != NULL) {
                free(data_out);
            }
            return ESP_ERR_NO_MEM;
        }
        int *tmp_data = NULL;
        if (data_out != NULL) {
            tmp_data = data_out;
        } else {
            tmp_data = (int *)data;
        }
        for (int i = 0; i < (int)(out_length / (audio_time * sizeof(int))); i++) {
            for (int j = 0; j < audio_time; j++) {
                data_out_1[audio_time * i + j] = tmp_data[i];
            }
        }
        if (data_out != NULL) {
            free(data_out);
            data_out = NULL;
        }
    }
    if (data_out != NULL) {
        ret = esp_codec_dev_write(play_dev, (void *)data_out, out_length);
        free(data_out);
    } else if (data_out_1 != NULL) {
        ret = esp_codec_dev_write(play_dev, (void *)data_out_1, out_length);
        free(data_out_1);
    } else {
        ret = esp_codec_dev_write(play_dev, (void *)data, length);
    }
    return ret;
}
esp_err_t esp_get_feed_data(bool is_get_raw_channel, int16_t *buffer, int buffer_len)
{
    esp_err_t ret = ESP_OK;
    size_t bytes_read;
    int audio_chunksize = buffer_len / (sizeof(int16_t) * ADC_I2S_CHANNEL);
    ret = esp_codec_dev_read(record_dev, (void *)buffer, buffer_len);
    if (!is_get_raw_channel) {
        for (int i = 0; i < audio_chunksize; i++) {
            int16_t ref = buffer[4 * i + 0];
            buffer[3 * i + 0] = buffer[4 * i + 1];
            buffer[3 * i + 1] = buffer[4 * i + 3];
            buffer[3 * i + 2] = ref;
        }
    }
    return ret;
}

esp_err_t esp_board_init(uint32_t sample_rate, int channel_format, int bits_per_chan)
{
    esp_err_t ret = i2c_master_init();
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "i2c_master_init failed: %s", esp_err_to_name(ret));
        return ret;
    }
    s_play_sample_rate = sample_rate;
    if (channel_format != 2 && channel_format != 1) {
        ESP_LOGE(TAG, "Unable to configure channel_format");
        channel_format = 2;
    }
    s_play_channel_format = channel_format;
    if (bits_per_chan != 32 && bits_per_chan != 16) {
        ESP_LOGE(TAG, "Unable to configure bits_per_chan");
        bits_per_chan = 32;
    }
    s_bits_per_chan = bits_per_chan;
    ret = bsp_i2s_init(I2S_NUM_1, (uint32_t)sample_rate, channel_format, bits_per_chan);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "bsp_i2s_init failed: %s", esp_err_to_name(ret));
        return ret;
    }
    ret = bsp_codec_init((int)sample_rate, (int)sample_rate, channel_format, bits_per_chan);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "bsp_codec_init failed: %s", esp_err_to_name(ret));
        return ret;
    }
    return ESP_OK;
}
esp_err_t esp_sdcard_init(char *mount_point, size_t max_files)
{
    if (NULL != card) {
        return ESP_ERR_INVALID_STATE;
    }
    if (!FUNC_SDMMC_EN && !FUNC_SDSPI_EN) {
        ESP_LOGE(TAG, "SDMMC and SDSPI not supported on this board!");
        return ESP_ERR_NOT_SUPPORTED;
    }
    esp_err_t ret_val = ESP_OK;
    esp_vfs_fat_sdmmc_mount_config_t mount_config = {
        .format_if_mount_failed = false,
        .max_files = max_files,
        .allocation_unit_size = 16 * 1024
    };
    sdmmc_host_t host =
#if FUNC_SDMMC_EN
        SDMMC_HOST_DEFAULT();
#else
        SDSPI_HOST_DEFAULT();
    spi_bus_config_t bus_cfg = {
        .mosi_io_num = GPIO_SDSPI_MOSI,
        .miso_io_num = GPIO_SDSPI_MISO,
        .sclk_io_num = GPIO_SDSPI_SCLK,
        .quadwp_io_num = GPIO_NUM_NC,
        .quadhd_io_num = GPIO_NUM_NC,
        .max_transfer_sz = 4000,
    };
    ret_val = spi_bus_initialize(host.slot, &bus_cfg, SPI_DMA_CH_AUTO);
    if (ret_val != ESP_OK) {
        ESP_LOGE(TAG, "Failed to initialize bus.");
        return ret_val;
    }
#endif
#if FUNC_SDMMC_EN
    sdmmc_slot_config_t slot_config = SDMMC_SLOT_CONFIG_DEFAULT();
#else
    sdspi_device_config_t slot_config = SDSPI_DEVICE_CONFIG_DEFAULT();
#endif
#if FUNC_SDMMC_EN
    slot_config.width = SDMMC_BUS_WIDTH;
#if SOC_SDMMC_USE_GPIO_MATRIX
    slot_config.clk = GPIO_SDMMC_CLK;
    slot_config.cmd = GPIO_SDMMC_CMD;
    slot_config.d0 = GPIO_SDMMC_D0;
    slot_config.d1 = GPIO_SDMMC_D1;
    slot_config.d2 = GPIO_SDMMC_D2;
    slot_config.d3 = GPIO_SDMMC_D3;
#endif
    slot_config.cd = GPIO_SDMMC_DET;
    slot_config.flags |= SDMMC_SLOT_FLAG_INTERNAL_PULLUP;
#else
    slot_config.gpio_cs = GPIO_SDSPI_CS;
    slot_config.host_id = host.slot;
#endif
    ret_val =
#if FUNC_SDMMC_EN
        esp_vfs_fat_sdmmc_mount(mount_point, &host, &slot_config, &mount_config, &card);
#else
        esp_vfs_fat_sdspi_mount(mount_point, &host, &slot_config, &mount_config, &card);
#endif
    if (ret_val != ESP_OK) {
        if (ret_val == ESP_FAIL) {
            ESP_LOGE(TAG, "Failed to mount filesystem. "
                     "If you want the card to be formatted, set the EXAMPLE_FORMAT_IF_MOUNT_FAILED menuconfig option.");
        } else {
            ESP_LOGE(TAG, "Failed to initialize the card (%s). "
                     "Make sure SD card lines have pull-up resistors in place.", esp_err_to_name(ret_val));
        }
        return ret_val;
    }
    sdmmc_card_print_info(stdout, card);
    SDCard_Size = ((uint64_t) card->csd.capacity) * card->csd.sector_size / (1024 * 1024);
    return ret_val;
}
esp_err_t esp_sdcard_deinit(char *mount_point)
{
    if (NULL == mount_point) {
        return ESP_ERR_INVALID_STATE;
    }
    esp_err_t ret_val = esp_vfs_fat_sdcard_unmount(mount_point, card);
    card = NULL;
    return ret_val;
}

#include "button_driver.h"
#include <atomic>
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_timer.h"
#include "tca9555_driver.h"
#include "general/input/button_state_machine.hpp"
static const char *TAG = "button_driver";
static general::input::ButtonStateMachine s_keys[3];
static key_callback_t s_callback = NULL;
static std::atomic<bool> s_running{false};
static std::atomic<bool> s_task_alive{false};
static void *s_user_data = NULL;
static bool get_key_state(uint32_t mask, key_id_t id) {
    return !(mask & (1ULL << id));
}
static void scan_timer_ccan(void *arg) {
    (void)arg;
    while (s_running.load()) {
        uint32_t mask;
        esp_err_t ret = esp_io_expander_get_level(
            io_expander,
            (1ULL << KEY_ID_9) | (1ULL << KEY_ID_10) | (1ULL << KEY_ID_11),
            &mask);
        if (ret != ESP_OK) {
            ESP_LOGE(TAG, "读取IO扩展芯片失败");
            vTaskDelay(pdMS_TO_TICKS(100));
            continue;
        }
        const uint32_t now = static_cast<uint32_t>(esp_timer_get_time() / 1000);
        for (int i = 0; i < 3; i++) {
            const key_id_t key_id = static_cast<key_id_t>(KEY_ID_9 + i);
            const bool pressed = get_key_state(mask, key_id);
            const general::input::ButtonEvent event = s_keys[i].update(pressed, now);
            if (event == general::input::ButtonEvent::ShortPress && s_callback) {
                s_callback(key_id, KEY_EVENT_SHORT_PRESS, s_user_data);
            } else if (event == general::input::ButtonEvent::LongPress && s_callback) {
                s_callback(key_id, KEY_EVENT_LONG_PRESS, s_user_data);
            }
        }
        vTaskDelay(pdMS_TO_TICKS(100));
    }
    s_task_alive.store(false);
    vTaskDelete(NULL);
}
esp_err_t key_module_init(void *user_data) {
    if (s_task_alive.load()) {
        return ESP_ERR_INVALID_STATE;
    }
    s_user_data = user_data;
    s_running.store(true);
    s_task_alive.store(true);
    BaseType_t res = xTaskCreatePinnedToCore(scan_timer_ccan, "key_scan", 4096,
                                             NULL, 3, NULL, 0);
    if (res != pdPASS) {
        s_task_alive.store(false);
        s_running.store(false);
        return ESP_FAIL;
    }
    return ESP_OK;
}
esp_err_t key_register_callback(key_callback_t callback) {
    if (!callback) return ESP_ERR_INVALID_ARG;
    s_callback = callback;
    return ESP_OK;
}
esp_err_t key_module_deinit(void) {
    s_running.store(false);
    for (int i = 0; i < 50 && s_task_alive.load(); ++i) {
        vTaskDelay(pdMS_TO_TICKS(10));
    }
    s_callback = NULL;
    return ESP_OK;
}

void _set_single_led_color(RGB_example_color_t color ,uint32_t led_index)
{
    rgb_color_t current_color = color_rgb_map[color];
    if (led_strip_set_pixel(led_strip, led_index, current_color.r, current_color.g, current_color.b) != ESP_OK) {
        ESP_LOGW(TAG, "led_strip_set_pixel failed");
    }
    if (led_strip_refresh(led_strip) != ESP_OK) {
        ESP_LOGW(TAG, "led_strip_refresh failed");
    }

void _example_playing(bool *reset)
{
    static uint16_t time_count = 0;
    static uint8_t led_num = 0;
    static bool clean_flag = 0;
    if(*reset)
    {
        time_count = 0;
        led_num = 0;
        *reset = 0;
        if (led_strip_clear(led_strip) != ESP_OK) {
            ESP_LOGW(TAG, "led_strip_clear failed");
        }
    }
    time_count++;
    if(time_count>=5)
    {
        time_count=0;
        if(clean_flag!=1)
        {
            _set_single_led_color(RGB_color,led_num++);
            if(led_num>=LED_STRIP_LED_COUNT)
            {
                led_num=0;
                clean_flag = 1;
            }
        }
        else
        {
            ESP_ERROR_CHECK(led_strip_clear(led_strip));
            clean_flag = 0;
        }
    }
}
void _example_esp_sr_rec()
{
    static uint16_t time_count = 0;
    static bool led_on_off = 0;
    time_count++;
    if(time_count>=2)
    {
        time_count=0;
        if (led_on_off)
        {
            for (int i = 0; i < LED_STRIP_LED_COUNT; i++)
            {
                _set_single_led_color(RGB_color,i);
            }
        }
        else
        {
            if (led_strip_clear(led_strip) != ESP_OK) {
                ESP_LOGW(TAG, "led_strip_clear failed");
            }
        }
        led_on_off = !led_on_off;
    }
}
void _RGB_Example(void *arg)
{
    static RGB_example_mode_t RGB_mode = RGB_MODE_IDLE;
    rgb_queue_data_t msg;
    bool reset_playing_color = 0;
    for (int i = 0; i < LED_STRIP_LED_COUNT; i++) {
        _set_single_led_color(RGB_color,i);
    }
    while (1)
    {
        BaseType_t ret = xQueueReceive(g_msg_queue, &msg, pdMS_TO_TICKS(100));
        if (ret == pdPASS)
        {
            switch (msg.cmd)
            {
                case RGB_CMD_SET_MODE:
                    RGB_mode = msg.param.mode;
                    reset_playing_color = 1;
                    if (led_strip_clear(led_strip) != ESP_OK) {
                        ESP_LOGW(TAG, "led_strip_clear failed");
                    }
                    if(RGB_mode == RGB_MODE_IDLE)
                    {
                        for (int i = 0; i < LED_STRIP_LED_COUNT; i++) {
                            _set_single_led_color(RGB_color,i);
                        }
                    }
                    break;
                case RGB_CMD_SET_COLOR:
                    RGB_color = msg.param.color;
                    reset_playing_color=1;
                    if (RGB_mode == RGB_MODE_IDLE) {
                        for (int i = 0; i < LED_STRIP_LED_COUNT; i++) {
                            _set_single_led_color(RGB_color, i);
                        }
                    }
                    break;
                default:
                    break;
            }
        }
        else
        {
            switch (RGB_mode)
            {
                case RGB_MODE_IDLE:
                    break;
                case RGB_MODE_PLAYING:
                    _example_playing(&reset_playing_color);
                    break;
                case RGB_MODE_REC_COMMAND:
                    _example_esp_sr_rec();
                    break;
                default:
                    break;
            }
        }
    }
}

void feed_Task(void *arg)
{
    esp_afe_sr_data_t *afe_data = arg;
    int audio_chunksize = afe_handle->get_feed_chunksize(afe_data);
    int nch = afe_handle->get_feed_channel_num(afe_data);
    int feed_channel = esp_get_feed_channel();
    assert(nch == feed_channel);
    int16_t *i2s_buff = heap_caps_malloc(audio_chunksize * sizeof(int16_t) * feed_channel,MALLOC_CAP_SPIRAM);
    assert(i2s_buff);
    esp_task_wdt_add(NULL);
    while (task_flag) {
        esp_get_feed_data(true, i2s_buff, audio_chunksize * sizeof(int16_t) * feed_channel);
        afe_handle->feed(afe_data, i2s_buff);
        esp_task_wdt_reset();
    }
    if (i2s_buff) {
        free(i2s_buff);
        i2s_buff = NULL;
    }
    vTaskDelete(NULL);
}
void detect_Task(void *arg)
{
    esp_afe_sr_data_t *afe_data = arg;
    int afe_chunksize = afe_handle->get_fetch_chunksize(afe_data);
    char *mn_name = esp_srmodel_filter(models, ESP_MN_PREFIX, ESP_MN_CHINESE);
    printf("multinet:%s\n", mn_name);
    esp_mn_iface_t *multinet = esp_mn_handle_from_name(mn_name);
    model_iface_data_t *model_data = multinet->create(mn_name, 6000);
    esp_mn_commands_update_from_sdkconfig(multinet, model_data);
    int mu_chunksize = multinet->get_samp_chunksize(model_data);
    assert(mu_chunksize == afe_chunksize);
    multinet->print_active_speech_commands(model_data);
    esp_task_wdt_add(NULL);
    while (task_flag) {
        afe_fetch_result_t* res = afe_handle->fetch(afe_data);
        if (!res || res->ret_value == ESP_FAIL) {
            printf("fetch error!\n");
            break;
        }
        if (res->wakeup_state == WAKENET_DETECTED) {
	        multinet->clean(model_data);
        }
        if (res->raw_data_channels == 1 && res->wakeup_state == WAKENET_DETECTED) {
            esp_sr_evt_data_t evtdata;
            evtdata.awaken_channel = 0;
            if (spench_callback != NULL) {
                spench_callback(ESP_SR_EVT_AWAKEN, evtdata, NULL);
            }
            wakeup_flag = 1;
        } else if (res->raw_data_channels > 1 && res->wakeup_state == WAKENET_CHANNEL_VERIFIED) {
            esp_sr_evt_data_t evtdata;
            evtdata.awaken_channel = res->trigger_channel_id;
            if(spench_callback!=NULL)
            {
                spench_callback(ESP_SR_EVT_AWAKEN,evtdata,NULL);
            }
            wakeup_flag = 1;
        }
        esp_task_wdt_reset();
        if (wakeup_flag == 1) {
            esp_mn_state_t mn_state = multinet->detect(model_data, res->data);
            if (mn_state == ESP_MN_STATE_DETECTING) {
                continue;
            }
            if (mn_state == ESP_MN_STATE_DETECTED) {
                esp_mn_results_t *mn_result = multinet->get_results(model_data);
                esp_sr_evt_data_t evtdata;
                evtdata.sr_cmd = mn_result->command_id[0];
                if(spench_callback!=NULL)
                {
                    spench_callback(ESP_SR_EVT_CMD,evtdata,NULL);
                }
            }
            if (mn_state == ESP_MN_STATE_TIMEOUT) {
                esp_mn_results_t *mn_result = multinet->get_results(model_data);
                afe_handle->enable_wakenet(afe_data);
                wakeup_flag = 0;
                esp_sr_evt_data_t evtdata;
                if(spench_callback!=NULL)
                {
                    spench_callback(ESP_SR_EVT_CMD_TIMEOUT,evtdata,NULL);
                }
                continue;
            }
        }
        esp_task_wdt_reset();
    }
    if (model_data) {
        multinet->destroy(model_data);
        model_data = NULL;
    }
    printf("detect exit\n");
    vTaskDelete(NULL);
}

esp_err_t init(const std::string& base_path, const std::string& partition_label, size_t max_files) {
    (void)max_files;
    s_base_path = base_path;
    s_partition_label = partition_label;
    ESP_LOGI(TAG, "Initializing LittleFS [%s] at [%s]", s_partition_label.c_str(), s_base_path.c_str());
    esp_vfs_littlefs_conf_t conf = {
        .base_path = s_base_path.c_str(),
        .partition_label = s_partition_label.c_str(),
        .format_if_mount_failed = true,
        .dont_mount = false
    };
    esp_err_t ret = esp_vfs_littlefs_register(&conf);
    if (ret != ESP_OK) {
        if (ret == ESP_FAIL) {
            ESP_LOGE(TAG, "Failed to mount or format filesystem");
        } else if (ret == ESP_ERR_NOT_FOUND) {
            ESP_LOGE(TAG, "Failed to find LittleFS partition '%s'", s_partition_label.c_str());
        } else {
            ESP_LOGE(TAG, "Failed to initialize LittleFS (%s)", esp_err_to_name(ret));
        }
        return ret;
    }
    s_is_mounted = true;
    return ESP_OK;
}
esp_err_t deinit() {
    if (!s_is_mounted) {
        return ESP_OK;
    }
    esp_err_t ret = esp_vfs_littlefs_unregister(s_partition_label.c_str());
    if (ret == ESP_OK) {
        s_is_mounted = false;
        ESP_LOGI(TAG, "LittleFS unmounted successfully");
    } else {
        ESP_LOGE(TAG, "Failed to unmount LittleFS (%s)", esp_err_to_name(ret));
    }
    return ret;
}
bool isMounted() {
    return s_is_mounted;
}
}

FirmwareSwitch::FirmwareSwitch()
    : m_running_partition(nullptr),
      m_next_partition(nullptr),
      m_ota_handle(0),
      m_is_initialized(false),
      m_is_updating(false) {}
FirmwareSwitch::~FirmwareSwitch() {
    if (m_is_updating && m_ota_handle != 0) {
        esp_ota_abort(m_ota_handle);
    }
}
bool FirmwareSwitch::init() {
    m_running_partition = esp_ota_get_running_partition();
    if (m_running_partition == nullptr) {
        ESP_LOGE(TAG, "Failed to get running firmware partition");
        return false;
    }
    m_next_partition = esp_ota_get_next_update_partition(nullptr);
    if (m_next_partition == nullptr) {
        ESP_LOGE(TAG, "Failed to get next update firmware partition");
        return false;
    }
    m_is_initialized = true;
    ESP_LOGI(TAG, "FirmwareSwitch initialized. Running: [%s], Alternate: [%s]",
             m_running_partition->label, m_next_partition->label);
    return true;
}
std::string FirmwareSwitch::getRunningPartitionName() const {
    if (!m_is_initialized) return "";
    return std::string(m_running_partition->label);
}
std::string FirmwareSwitch::getNextPartitionName() const {
    if (!m_is_initialized) return "";
    return std::string(m_next_partition->label);
}
bool FirmwareSwitch::switchToAlternateSlot() {
    if (!m_is_initialized) return false;
    ESP_LOGI(TAG, "Switching boot partition to alternate slot: [%s]", m_next_partition->label);
    esp_err_t ret = esp_ota_set_boot_partition(m_next_partition);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Failed to set boot partition: %s", esp_err_to_name(ret));
        return false;
    }
    ESP_LOGI(TAG, "Boot partition updated successfully. Restart required to apply.");
    return true;
}
bool FirmwareSwitch::markAppValid() {
    if (!m_is_initialized) return false;
    esp_err_t ret = esp_ota_mark_app_valid_cancel_rollback();
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Failed to mark application as valid: %s", esp_err_to_name(ret));
        return false;
    }
    ESP_LOGI(TAG, "Application marked as valid and rollback cancelled successfully.");
    return true;
}
bool FirmwareSwitch::beginOtaUpdate(size_t image_size) {
    if (!m_is_initialized) return false;
    if (m_is_updating) {
        ESP_LOGW(TAG, "OTA update session is already in progress.");
        return false;
    }
    ESP_LOGI(TAG, "Beginning OTA update session on alternate partition: [%s]", m_next_partition->label);
    esp_err_t ret = esp_ota_begin(m_next_partition, image_size, &m_ota_handle);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "esp_ota_begin failed: %s", esp_err_to_name(ret));
        return false;
    }
    m_is_updating = true;
    ESP_LOGI(TAG, "OTA begin successful. Ready to receive firmware data chunks.");
    return true;
}
bool FirmwareSwitch::writeOtaData(const void* data, size_t size) {
    if (!m_is_initialized || !m_is_updating) return false;
    esp_err_t ret = esp_ota_write(m_ota_handle, (const void*)data, size);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "esp_ota_write failed: %s; aborting OTA session", esp_err_to_name(ret));
        esp_ota_abort(m_ota_handle);
        m_ota_handle = 0;
        m_is_updating = false;
        return false;
    }
    return true;
}
bool FirmwareSwitch::endOtaUpdate() {
    if (!m_is_initialized || !m_is_updating) return false;
    esp_err_t ret = esp_ota_end(m_ota_handle);
    m_is_updating = false;
    m_ota_handle = 0;
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "esp_ota_end failed: %s", esp_err_to_name(ret));
        return false;
    }
    ESP_LOGI(TAG, "OTA update written successfully. Setting alternate slot as boot target...");
    return switchToAlternateSlot();
}

#include "crypto.hpp"
#include <cstring>
namespace encryption {
bool aes_gcm_decrypt(const uint8_t key[AES_KEY_BYTES],
                     const uint8_t* packet, size_t packet_len,
                     uint8_t* plaintext_out, size_t* plaintext_len,
                     CryptoBackend& backend)
{
    if (!backend.init()) {
        return false;
    }
    if (packet_len < GCM_NONCE_BYTES + GCM_TAG_BYTES) {
        LOG_E(TAG, "Packet too short");
        return false;
    }
    const size_t ciphertext_len = packet_len - GCM_NONCE_BYTES - GCM_TAG_BYTES;
    if (*plaintext_len < ciphertext_len) {
        LOG_E(TAG, "Output buffer too small");
        return false;
    }
    const uint8_t* nonce = packet;
    const uint8_t* tag = packet + GCM_NONCE_BYTES;
    const uint8_t* ciphertext = packet + GCM_NONCE_BYTES + GCM_TAG_BYTES;
    psa_key_attributes_t attributes = PSA_KEY_ATTRIBUTES_INIT;
    psa_set_key_usage_flags(&attributes, PSA_KEY_USAGE_DECRYPT);
    psa_set_key_algorithm(&attributes, PSA_ALG_GCM);
    psa_set_key_type(&attributes, PSA_KEY_TYPE_AES);
    psa_set_key_bits(&attributes, AES_KEY_BYTES * 8);
    psa_key_id_t key_id;
    psa_status_t status = psa_import_key(&attributes, key, AES_KEY_BYTES, &key_id);
    if (status != PSA_SUCCESS) {
        LOG_E(TAG, "Failed to import key: %d", static_cast<int>(status));
        return false;
    }
    std::vector<uint8_t> combined(ciphertext_len + GCM_TAG_BYTES);
    std::memcpy(combined.data(), ciphertext, ciphertext_len);
    std::memcpy(combined.data() + ciphertext_len, tag, GCM_TAG_BYTES);
    size_t decrypted_size = 0;
    status = psa_aead_decrypt(
        key_id,
        PSA_ALG_GCM,
        nonce, GCM_NONCE_BYTES,
        nullptr, 0,
        combined.data(), combined.size(),
        plaintext_out, *plaintext_len,
        &decrypted_size);
    psa_destroy_key(key_id);
    if (status != PSA_SUCCESS) {
        LOG_E(TAG, "Decryption or auth tag verification failed! Error: %d",
              static_cast<int>(status));
        return false;
    }
    *plaintext_len = decrypted_size;
    return true;
}
std::optional<std::vector<uint8_t>> aes_gcm_decrypt(const std::vector<uint8_t>& packet,
                                                    const std::vector<uint8_t>& key,
                                                    CryptoBackend& backend)
{
    if (key.size() != AES_KEY_BYTES) {
        LOG_E(TAG, "Invalid key size: expected %u, got %u",
              static_cast<unsigned>(AES_KEY_BYTES),
              static_cast<unsigned>(key.size()));
        return std::nullopt;
    }
    if (packet.size() < GCM_NONCE_BYTES + GCM_TAG_BYTES) {
        LOG_E(TAG, "Packet too short");
        return std::nullopt;
    }
    size_t plaintext_len = packet.size() - GCM_NONCE_BYTES - GCM_TAG_BYTES;
    std::vector<uint8_t> plaintext_out(plaintext_len);
    const bool ok = aes_gcm_decrypt(key.data(), packet.data(), packet.size(),
                                    plaintext_out.data(), &plaintext_len, backend);
    if (!ok) {
        return std::nullopt;
    }
    plaintext_out.resize(plaintext_len);
    return plaintext_out;
}

#include "crypto.hpp"
#include <cstring>
namespace encryption {
bool aes_gcm_encrypt(const uint8_t key[AES_KEY_BYTES],
                     const uint8_t* plaintext, size_t len,
                     uint8_t* out_buffer, size_t* out_len,
                     CryptoBackend& backend)
{
    if (!backend.init()) {
        return false;
    }
    if (*out_len < len + GCM_NONCE_BYTES + GCM_TAG_BYTES) {
        LOG_E(TAG, "Buffer capacity too small");
        return false;
    }
    psa_key_attributes_t attributes = PSA_KEY_ATTRIBUTES_INIT;
    psa_set_key_usage_flags(&attributes, PSA_KEY_USAGE_ENCRYPT);
    psa_set_key_algorithm(&attributes, PSA_ALG_GCM);
    psa_set_key_type(&attributes, PSA_KEY_TYPE_AES);
    psa_set_key_bits(&attributes, AES_KEY_BYTES * 8);
    psa_key_id_t key_id;
    psa_status_t status = psa_import_key(&attributes, key, AES_KEY_BYTES, &key_id);
    if (status != PSA_SUCCESS) {
        LOG_E(TAG, "Failed to import key: %d", static_cast<int>(status));
        return false;
    }
    uint8_t nonce[GCM_NONCE_BYTES];
    backend.fillRandom(nonce, GCM_NONCE_BYTES);
    std::vector<uint8_t> encrypted_payload(len + GCM_TAG_BYTES);
    size_t encrypted_output_len = 0;
    status = psa_aead_encrypt(
        key_id,
        PSA_ALG_GCM,
        nonce, GCM_NONCE_BYTES,
        nullptr, 0,
        plaintext, len,
        encrypted_payload.data(), encrypted_payload.size(),
        &encrypted_output_len);
    psa_destroy_key(key_id);
    if (status != PSA_SUCCESS) {
        LOG_E(TAG, "Encryption failed: %d", static_cast<int>(status));
        return false;
    }
    std::memcpy(out_buffer, nonce, GCM_NONCE_BYTES);
    std::memcpy(out_buffer + GCM_NONCE_BYTES, encrypted_payload.data() + len, GCM_TAG_BYTES);
    std::memcpy(out_buffer + GCM_NONCE_BYTES + GCM_TAG_BYTES, encrypted_payload.data(), len);
    *out_len = len + GCM_NONCE_BYTES + GCM_TAG_BYTES;
    return true;
}
std::optional<std::vector<uint8_t>> aes_gcm_encrypt(
    const std::vector<uint8_t>& plaintext, const std::vector<uint8_t>& aes_key,
    CryptoBackend& backend)
{
    if (aes_key.size() != AES_KEY_BYTES) {
        LOG_E(TAG, "Invalid key size: expected %u, got %u",
              static_cast<unsigned>(AES_KEY_BYTES),
              static_cast<unsigned>(aes_key.size()));
        return std::nullopt;
    }
    size_t out_len = plaintext.size() + GCM_NONCE_BYTES + GCM_TAG_BYTES;
    std::vector<uint8_t> out_buffer(out_len);
    const bool ok = aes_gcm_encrypt(aes_key.data(), plaintext.data(), plaintext.size(),
                                    out_buffer.data(), &out_len, backend);
    if (!ok) {
        return std::nullopt;
    }
    out_buffer.resize(out_len);
    return out_buffer;
}

namespace general::fs {
std::vector<std::string> listFiles(const std::string& directory,
                                   const std::string& extension,
                                   std::size_t max_files) {
    std::vector<std::string> files;
    DIR* dir = ::opendir(directory.c_str());
    if (dir == nullptr) {
        return files;
    }
    struct dirent* entry;
    while ((entry = ::readdir(dir)) != nullptr) {
        if (max_files != 0 && files.size() >= max_files) {
            break;
        }
        if (std::strcmp(entry->d_name, ".") == 0 || std::strcmp(entry->d_name, "..") == 0) {
            continue;
        }
        const char* dot = std::strrchr(entry->d_name, '.');
        if (dot == nullptr || dot == entry->d_name) {
            continue;
        }
        if (::strcasecmp(dot, extension.c_str()) != 0) {
            continue;
        }
        files.emplace_back(entry->d_name);
    }
    ::closedir(dir);
    return files;
}

ButtonEvent ButtonStateMachine::update(bool pressed, uint32_t now_ms) {
    ButtonEvent event = ButtonEvent::None;
    pressed_ = pressed;
    if (pressed_ && !last_pressed_) {
        press_start_ms_ = now_ms;
        long_triggered_ = false;
    } else if (!pressed_ && last_pressed_) {
        if (!long_triggered_) {
            const uint32_t press_time = now_ms - press_start_ms_;
            event = (press_time >= long_press_ms_)
                        ? ButtonEvent::LongPress
                        : ButtonEvent::ShortPress;
        }
    } else if (pressed_ && !long_triggered_) {
        if (now_ms - press_start_ms_ >= long_press_ms_) {
            event = ButtonEvent::LongPress;
            long_triggered_ = true;
        }
    }
    last_pressed_ = pressed_;
    return event;
}

    fflush(m_file);
    long currentPos = ftell(m_file);
    if (fseek(m_file, 0, SEEK_END) != 0) {
        LOG_E(TAG, "fseek(SEEK_END) failed on '%s' (errno %d: %s)", m_fullPath.c_str(), errno, strerror(errno));
        return 0;

static bool writeRaw(FILE* file, const uint8_t* dataPtr, size_t totalBytes, long offset, RandomAccessFile* self) {
    long currentSize = static_cast<long>(self->fileSize());
    if (offset < 0) {
        offset = currentSize;
    }
    if (offset > currentSize) {
        if (!self->seek(0, SEEK_END)) {
            LOG_E(TAG, "Failed seeking to EOF for padding (errno %d: %s)", errno, strerror(errno));
            return false;
        }
        long gap = offset - currentSize;
        constexpr size_t ZERO_BUF_SIZE = 256;
        const uint8_t zeroBuffer[ZERO_BUF_SIZE] = {0};
        while (gap > 0) {
            size_t toWrite = std::min(static_cast<size_t>(gap), ZERO_BUF_SIZE);
            size_t written = fwrite(zeroBuffer, 1, toWrite, file);
            if (written != toWrite) {
                LOG_E(TAG, "Failed to write zero padding: wrote %zu of %zu (errno %d: %s)", written, toWrite, errno, strerror(errno));
                return false;
            }
            gap -= toWrite;
        }
    } else {
        if (!self->seek(offset, SEEK_SET)) {
            LOG_E(TAG, "Failed seeking to offset %ld (errno %d: %s)", offset, errno, strerror(errno));
            return false;
        }
    }
    if (totalBytes > 0 && dataPtr != nullptr) {
        size_t bytesWrittenTotal = 0;
        constexpr size_t CHUNK_SIZE = 4096;
        while (bytesWrittenTotal < totalBytes) {
            size_t chunkBytes = std::min(CHUNK_SIZE, totalBytes - bytesWrittenTotal);
            size_t written = fwrite(dataPtr + bytesWrittenTotal, 1, chunkBytes, file);
            if (written != chunkBytes) {
                LOG_E(TAG, "Short write: wrote %zu of %zu bytes (total written: %zu, errno %d: %s)",
                         written, chunkBytes, bytesWrittenTotal, errno, strerror(errno));
                return false;
            }
            bytesWrittenTotal += written;
        }
    }
    return true;
}
bool RandomAccessFile::writeFile(const std::string& data, long offset) {
    if (!openInternal()) return false;
    return writeRaw(m_file, reinterpret_cast<const uint8_t*>(data.data()), data.size(), offset, this);
}
bool RandomAccessFile::writeFile(const std::vector<uint8_t>& data, long offset) {
    if (!openInternal()) return false;
    return writeRaw(m_file, data.data(), data.size(), offset, this);
}
std::vector<uint8_t> RandomAccessFile::readFileBytes(long offset, long size) {
    if (!openInternal()) return {};
    long totalLength = static_cast<long>(fileSize());
    if (totalLength <= 0) {
        return {};
    }
    if (offset < 0 || offset >= totalLength) {
        LOG_W(TAG, "Read offset %ld out of bounds for '%s' (file size: %ld)", offset, m_fullPath.c_str(), totalLength);
        return {};
    }
    size_t bytesToRead = 0;
    if (size < 0) {
        bytesToRead = static_cast<size_t>(totalLength - offset);
    } else {
        bytesToRead = std::min(static_cast<size_t>(size), static_cast<size_t>(totalLength - offset));
    }
    if (bytesToRead == 0) {
        return {};
    }
    if (!seek(offset, SEEK_SET)) {
        LOG_E(TAG, "Failed seeking to offset %ld in '%s' (errno %d: %s)", offset, m_fullPath.c_str(), errno, strerror(errno));
        return {};
    }
    std::vector<uint8_t> result(bytesToRead);
    size_t readCount = fread(result.data(), 1, bytesToRead, m_file);
    if (readCount != bytesToRead) {
        LOG_W(TAG, "Short read on '%s': expected %zu bytes, read %zu bytes (errno %d: %s)",
                 m_fullPath.c_str(), bytesToRead, readCount, errno, strerror(errno));
        result.resize(readCount);
    }
    return result;
}
void RandomAccessFile::shiftBytes(size_t shiftDistance, bool left, long long startOffset, long long endOffset) {
    if (shiftDistance == 0) return;
    long currentFileSize = static_cast<long>(fileSize());
    if (currentFileSize <= 0) return;
    long actualStart = (startOffset < 0) ? 0 : static_cast<long>(startOffset);
    long actualEnd = (endOffset < 0 || endOffset > currentFileSize) ? currentFileSize : static_cast<long>(endOffset);
    if (actualStart >= currentFileSize || actualStart >= actualEnd) return;
    if (left && static_cast<long>(shiftDistance) > actualStart) {
        LOG_E(TAG, "Cannot shift left: underflow offset 0");
        return;
    }
    if (m_file) {
        fclose(m_file);
        m_file = nullptr;
    }
    std::string filePath = m_fullPath;
    std::string tmpPath = filePath + ".tmp";
    FILE* src = fopen(filePath.c_str(), "rb");
    FILE* dst = fopen(tmpPath.c_str(), "wb");
    if (!src || !dst) {
                LOG_E(TAG, "shiftBytes fopen failed! src: %p, dst: %p, path: '%s' (errno %d: %s)",
                 src, dst, filePath.c_str(), errno, strerror(errno));
        if (src) fclose(src);
        if (dst) fclose(dst);
        unlink(tmpPath.c_str());
        openInternal();
        return;
    }
    constexpr size_t BUF_SIZE = 4096;
    auto buf = std::unique_ptr<uint8_t[]>(new (std::nothrow) uint8_t[BUF_SIZE]);
    if (!buf) {
        fclose(src);
        fclose(dst);
        unlink(tmpPath.c_str());
        openInternal();
        return;
    }
    bool write_failed = false;
    auto copyRange = [&](long from, long to) {
        fseek(src, from, SEEK_SET);
        long remaining = to - from;
        while (remaining > 0 && !write_failed) {
            size_t n = std::min<size_t>(remaining, BUF_SIZE);
            size_t r = fread(buf.get(), 1, n, src);
            if (r == 0) {
                write_failed = true;
                break;
            }
            size_t w = fwrite(buf.get(), 1, r, dst);
            if (w != r) {
                write_failed = true;
                break;
            }
            remaining -= r;
        }
    };
    if (left) {
        if (actualStart > static_cast<long>(shiftDistance)) {
            copyRange(0, actualStart - shiftDistance);
        }
        copyRange(actualStart, actualEnd);
        if (actualEnd < currentFileSize) {
            copyRange(actualEnd, currentFileSize);
        }
    } else {
        if (actualStart > 0) {
            copyRange(0, actualStart);
        }
        std::vector<uint8_t> zeros(std::min<size_t>(shiftDistance, BUF_SIZE), 0x00);
        size_t holeLeft = shiftDistance;
        while (holeLeft > 0 && !write_failed) {
            size_t n = std::min(holeLeft, zeros.size());
            size_t w = fwrite(zeros.data(), 1, n, dst);
            if (w != n) {
                write_failed = true;
                break;
            }
            holeLeft -= n;
        }
        copyRange(actualStart, actualEnd);
        if (actualEnd < currentFileSize) {
            copyRange(actualEnd, currentFileSize);
        }
    }
    fflush(dst);
    if (ferror(dst)) {
        write_failed = true;
    }
    fclose(src);
    fclose(dst);
    if (write_failed) {
        LOG_E(TAG, "shiftBytes write failed; keeping original '%s'", filePath.c_str());
        unlink(tmpPath.c_str());
        openInternal();
        return;
    }
    if (rename(tmpPath.c_str(), filePath.c_str()) != 0) {
        LOG_E(TAG, "shiftBytes rename failed: '%s' (errno %d: %s)",
              filePath.c_str(), errno, strerror(errno));
        unlink(tmpPath.c_str());
        openInternal();
        return;
    }
    openInternal();
}
bool RandomAccessFile::truncate(size_t newSize) {
    if (!openInternal()) return false;
    fflush(m_file);
    if (ftruncate(fileno(m_file), newSize) != 0) {
        LOG_E(TAG, "ftruncate to size %zu failed on '%s' (errno %d: %s)", newSize, m_fullPath.c_str(), errno, strerror(errno));
        return false;
    }
    return true;
}
bool RandomAccessFile::deleteFile() {
    if (m_file) {
        fclose(m_file);
        m_file = nullptr;
    }
    return ::unlink(m_fullPath.c_str()) == 0;
}
void RandomAccessFile::flush() {
    if (m_file) {
        fflush(m_file);
    }
}

bool StdioFileStore::readAll(const std::string& full_path, std::vector<uint8_t>& out) {
    FILE* f = std::fopen(full_path.c_str(), "rb");
    if (f == nullptr) {
        return false;
    }
    std::fseek(f, 0, SEEK_END);
    const long size = std::ftell(f);
    std::rewind(f);
    if (size < 0) {
        std::fclose(f);
        return false;
    }
    out.resize(static_cast<std::size_t>(size));
    const std::size_t read = out.empty() ? 0 : std::fread(out.data(), 1, out.size(), f);
    std::fclose(f);
    return read == out.size();
}
bool StdioFileStore::writeAll(const std::string& full_path, const std::vector<uint8_t>& data) {
    FILE* f = std::fopen(full_path.c_str(), "wb");
    if (f == nullptr) {
        return false;
    }
    const std::size_t written = data.empty() ? 0 : std::fwrite(data.data(), 1, data.size(), f);
    std::fclose(f);
    return written == data.size();
}

#include "general/kcp/kcp_session.hpp"
namespace general::kcp {
KcpSession::KcpSession(DatagramIo& io, const KcpConfig& config)
    : io_(io), config_(config) {}
KcpSession::~KcpSession() {
    close();
}
bool KcpSession::open() {
    if (kcp_ != nullptr) {
        return true;
    }
    kcp_ = ikcp_create(config_.conversation_id, this);
    if (kcp_ == nullptr) {
        return false;
    }
    ikcp_setoutput(kcp_, &KcpSession::outputCallback);
    ikcp_nodelay(kcp_, config_.nodelay, config_.interval_ms,
                 config_.resend, config_.no_congestion);
    ikcp_wndsize(kcp_, config_.send_window, config_.receive_window);
    return true;
}
void KcpSession::close() {
    if (kcp_ != nullptr) {
        ikcp_release(kcp_);
        kcp_ = nullptr;
    }
}
int KcpSession::outputCallback(const char* buf, int len, ikcpcb* , void* user) {
    auto* self = static_cast<KcpSession*>(user);
    if (self == nullptr || buf == nullptr || len <= 0) {
        return -1;
    }
    return self->io_.send(reinterpret_cast<const uint8_t*>(buf),
                          static_cast<std::size_t>(len));
}
int KcpSession::send(const uint8_t* data, std::size_t size) {
    if (kcp_ == nullptr || data == nullptr || size == 0) {
        return -1;
    }
    return ikcp_send(kcp_, reinterpret_cast<const char*>(data),
                     static_cast<int>(size));
}
std::vector<uint8_t> KcpSession::poll(uint32_t now_ms) {
    if (kcp_ == nullptr) {
        return {};
    }
    uint8_t rx_buffer[1500];
    constexpr int kMaxDatagramsPerPoll = 32;
    for (int datagrams = 0; datagrams < kMaxDatagramsPerPoll; ++datagrams) {
        const int received = io_.receive(rx_buffer, sizeof(rx_buffer));
        if (received <= 0) {
            break;
        }
        const int err = ikcp_input(kcp_, reinterpret_cast<const char*>(rx_buffer), received);
        if (err >= 0) {
            last_rx_ms_ = now_ms;
        }
    }
    const int peek_size = ikcp_peeksize(kcp_);
    if (peek_size <= 0) {
        return {};
    }
    std::vector<uint8_t> buffer(static_cast<std::size_t>(peek_size));
    const int bytes_read = ikcp_recv(kcp_, reinterpret_cast<char*>(buffer.data()),
                                     static_cast<int>(buffer.size()));
    if (bytes_read < 0) {
        return {};
    }
    buffer.resize(static_cast<std::size_t>(bytes_read));
    return buffer;
}
bool KcpSession::isDead() const {
    return kcp_ != nullptr && kcp_->state == static_cast<IUINT32>(-1);
}
void KcpSession::update(uint32_t now_ms) {
    if (kcp_ != nullptr) {
        ikcp_update(kcp_, now_ms);
    }
}

namespace general::net {
inline constexpr const char* TAG = "UDP";
struct UdpConnection {
    int sock = -1;
    struct sockaddr_storage addr = {};
    socklen_t addr_len = 0;
    bool is_valid() const { return sock >= 0; }
};
inline bool makeSocketNonBlocking(int sock) {
    const int flags = fcntl(sock, F_GETFL, 0);
    if (flags < 0) {
        LOG_E(TAG, "Failed to get socket flags: %d", errno);
        return false;
    }
    if (fcntl(sock, F_SETFL, flags | O_NONBLOCK) < 0) {
        LOG_E(TAG, "Failed to set socket to non-blocking: %d", errno);
        return false;
    }
    return true;
}
inline std::optional<UdpConnection> openUdpConnection(const char* host, const char* port) {
    struct addrinfo hints = {};
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_DGRAM;
    hints.ai_protocol = IPPROTO_UDP;
    struct addrinfo* res = nullptr;
    const int err = getaddrinfo(host, port, &hints, &res);
    if (err != 0 || res == nullptr) {
        LOG_E(TAG, "Failed to resolve hostname %s: error %d", host, err);
        return std::nullopt;
    }
    const int sock = ::socket(res->ai_family, res->ai_socktype, res->ai_protocol);
    if (sock < 0) {
        LOG_E(TAG, "Failed to create socket for family %d", res->ai_family);
        freeaddrinfo(res);
        return std::nullopt;
    }
    if (!makeSocketNonBlocking(sock)) {
        ::close(sock);
        freeaddrinfo(res);
        return std::nullopt;
    }
    UdpConnection conn;
    conn.sock = sock;
    std::memcpy(&conn.addr, res->ai_addr, res->ai_addrlen);
    conn.addr_len = res->ai_addrlen;
    if (::connect(sock, reinterpret_cast<const struct sockaddr*>(&conn.addr),
                  conn.addr_len) != 0) {
        LOG_E(TAG, "Failed to connect UDP socket: %d", errno);
        ::close(sock);
        freeaddrinfo(res);
        return std::nullopt;
    }
    freeaddrinfo(res);
    return conn;
}
inline int udpSend(const UdpConnection& conn, const uint8_t* data, size_t len) {
    if (!conn.is_valid() || data == nullptr || len == 0) {
        LOG_E(TAG, "Invalid parameters for send");
        return -1;
    }
    ssize_t sent;
    do {
        sent = sendto(conn.sock, data, len, 0,
                      reinterpret_cast<const struct sockaddr*>(&conn.addr),
                      conn.addr_len);
    } while (sent < 0 && errno == EINTR);
    if (sent < 0) {
        return -1;
    }
    return (static_cast<size_t>(sent) == len) ? 0 : -1;
}
inline int udpReceive(int sock, uint8_t* buffer, size_t capacity,
                      struct sockaddr_storage* src_addr, socklen_t* src_addr_len) {
    if (sock < 0 || buffer == nullptr || capacity == 0 ||
        src_addr == nullptr || src_addr_len == nullptr) {
        return -1;
    }
    ssize_t received;
    do {
        received = recvfrom(sock, buffer, capacity, 0,
                            reinterpret_cast<struct sockaddr*>(src_addr),
                            src_addr_len);
    } while (received < 0 && errno == EINTR);
    if (received < 0) {
        return (errno == EWOULDBLOCK || errno == EAGAIN) ? 0 : -1;
    }
    return static_cast<int>(received);
}

#!/usr/bin/env python3
import os
from pathlib import Path
SOURCE_EXTENSIONS = {
    ".cpp", ".hpp", ".h", ".cc", ".c",
    ".cxx", ".hxx", ".hh"
}
SPECIAL_FILES = {
    "CMakeLists.txt",
    "README.md",
    "Makefile"
}
EXCLUDED_DIRS = {
    ".git",
    ".vscode",
    ".idea",
    "__pycache__",
    "build",
    "dist",
    "venv",
    ".venv",
    "node_modules",
}
def should_include(path):
    return (
        path.suffix.lower() in SOURCE_EXTENSIONS
        or path.name in SPECIAL_FILES
    )
def get_files(root):
    files = []
    for current, dirs, filenames in os.walk(root):
        dirs[:] = sorted(
            d for d in dirs if d not in EXCLUDED_DIRS
        )
        for filename in sorted(filenames):
            path = Path(current) / filename
            if should_include(path):
                files.append(path)
    return sorted(files)
def get_tree(root, files):
    lines = [root.name + "/"]
    included_dirs = set()
    included_files = set(files)
    for file in files:
        parent = file.parent
        while parent != root:
            included_dirs.add(parent)
            parent = parent.parent
    all_paths = sorted(
        included_dirs | included_files,
        key=lambda p: (str(p.relative_to(root)).lower(),
                       p.is_file())
    )
    for path in all_paths:
        relative = path.relative_to(root)
        depth = len(relative.parts)
        name = relative.name
        suffix = "/" if path.is_dir() else ""
        lines.append("    " * depth + name + suffix)
    return "\n".join(lines)
def get_includes(files, root):
    result = []
    for file in files:
        try:
            content = file.read_text(
                encoding="utf-8",
                errors="replace"
            )
        except OSError:
            continue
        for line_number, line in enumerate(
            content.splitlines(), 1
        ):
            stripped = line.strip()
            if stripped.startswith("#include"):
                relative = file.relative_to(root)
                result.append(
                    f"{relative}:{line_number}: {stripped}"
                )
    return result
def export_project(root):
    files = get_files(root)
    output = root / "project_context.txt"
    with output.open("w", encoding="utf-8") as out:
        out.write("=" * 80 + "\n")
        out.write("C++ PROJECT CONTEXT\n")
        out.write("=" * 80 + "\n\n")
        out.write("PROJECT STRUCTURE\n")
        out.write("-" * 80 + "\n")
        out.write(get_tree(root, files))
        out.write("\n\n")
        out.write("INCLUDE RELATIONSHIPS\n")
        out.write("-" * 80 + "\n")
        includes = get_includes(files, root)
        if includes:
            out.write("\n".join(includes))
        else:
            out.write("No include directives found.")
        out.write("\n\n")
        out.write("SOURCE FILE CONTENTS\n")
        out.write("=" * 80 + "\n")
        for file in files:
            relative = file.relative_to(root)
            out.write("\n\n")
            out.write("=" * 80 + "\n")
            out.write(f"FILE: {relative}\n")
            out.write("=" * 80 + "\n\n")
            try:
                content = file.read_text(
                    encoding="utf-8",
                    errors="replace"
                )
                out.write(content)
            except OSError as error:
                out.write(f"[ERROR READING FILE: {error}]")
            out.write("\n")
    print("\nExport completed!")
    print(f"Files collected: {len(files)}")
    print(f"Output: {output}")

set -u
cd "$(dirname "$0")/.."
fail=0
PATTERN_APP='esp_[a-z]|freertos/|driver/|nvs_|nvs\.h|wifi::|psa/|esp_ota'
PATTERN_GENERAL='esp_[a-z]|freertos/|driver/|nvs_|nvs\.h|wifi::|esp_ota'
check_pure() {
    local dir="$1" label="$2" pattern="$3"
    [ -d "$dir" ] || return 0
    local hits
    hits=$(grep -rnE "$pattern" "$dir" \
               --include='*.cpp' --include='*.hpp' --include='*.h' 2>/dev/null \
           | grep -vE '^[^:]+:[0-9]+:[[:space:]]*(//|\*|/\*)' || true)
    if [ -n "$hits" ]; then
        echo "[FAIL] $label has platform leaks:"
        echo "$hits" | sed 's/^/       /'
        fail=1
    else
        echo "[ok]   $label"
    fi
}
check_pure main/core    "core     (pure C++)"  "$PATTERN_APP"
check_pure main/general "general  (portable)"  "$PATTERN_GENERAL"
if [ -d main/general ]; then
    hits=$(grep -rnE '#include[[:space:]]*"(interface\.hpp|core/|esp_hardware/)' main/general \
               --include='*.cpp' --include='*.hpp' --include='*.h' 2>/dev/null || true)
    if [ -n "$hits" ]; then
        echo "[FAIL] general depends on interface/core/esp_hardware:"
        echo "$hits" | sed 's/^/       /'
        fail=1
    fi
fi
hits=$(grep -rnE '#include[[:space:]]*"(app_context\.hpp|esp_hardware/|adapters/)' main/core \
           --include='*.cpp' --include='*.hpp' --include='*.h' 2>/dev/null || true)
if [ -n "$hits" ]; then
    echo "[FAIL] core depends on the composition root / hardware layer:"
    echo "$hits" | sed 's/^/       /'
    fail=1
else
    echo "[ok]   core     (no composition/hardware includes)"
fi
hits=$(grep -rnE '#include[[:space:]]*"core/' main/esp_hardware \
           --include='*.cpp' --include='*.hpp' --include='*.h' 2>/dev/null || true)
if [ -n "$hits" ]; then
    echo "[FAIL] esp_hardware depends on core/ (use interface.hpp or general/):"
    echo "$hits" | sed 's/^/       /'
    fail=1
else
    echo "[ok]   esp_hardware (no core/ includes)"

#include <cstdint>
#include <cstdio>
#include <deque>
#include <optional>
#include <string>
#include <vector>
#include "core/audio/AudioPipeline.hpp"
#include "core/session/SessionManager.hpp"

namespace {
int g_failures = 0;
#define CHECK(cond)                                                     \
    do {                                                                \
        if (!(cond)) {                                                  \
            std::printf("FAIL %s:%d: %s\n", __FILE__, __LINE__, #cond); \
            ++g_failures;                                               \
        }                                                               \
    } while (0)
struct FakeClock : core::IClock {
    uint32_t ms = 1000;
    uint32_t nowMs() const override { return ms; }
    int64_t nowUnixMs() const override { return static_cast<int64_t>(ms); }
    void advance(uint32_t d) { ms += d; }
};
struct FakeLogger : core::ILogger {
    void log(core::LogLevel, const char*) override {}
};
struct FakeRandom : core::IRandom {
    uint32_t values[4] = {0x11111111u, 0x22222222u, 0x33333333u, 0x44444444u};
    int i = 0;
    uint32_t u32() override { return values[i++ % 4]; }
};
struct FakeCrypto : core::ICrypto {
    std::optional<core::Bytes> generatePublicKey() override {
        return core::Bytes{0xAA, 0xBB, 0xCC};
    }
    std::optional<core::Bytes> computeSharedSecret(const core::Bytes&) override {
        return core::Bytes(32, 0x42);
    }
    std::optional<core::Bytes> encrypt(const core::Bytes& p, const core::Bytes&) override {
        return p;
    }
    std::optional<core::Bytes> decrypt(const core::Bytes& p, const core::Bytes&) override {
        return p;
    }
    std::optional<core::Bytes> sha256(const core::Bytes&) override {
        return core::Bytes(32, 0x5A);
    }
    std::optional<core::Bytes> hmacSha256(const core::Bytes&, const core::Bytes&) override {
        return core::Bytes(32, 0xA5);
    }
};
struct FakeLink : core::ILink {
    bool up = false;
    bool start() override { up = true; return true; }
    bool isUp() const override { return up; }
};
struct FakeTransport : core::ITransport {
    bool open_ = false;
    bool dead = false;
    uint32_t conv = 0;
    uint32_t last_rx = 0;
    std::deque<core::Bytes> incoming;
    std::vector<core::Bytes> raw_sent;
    std::vector<core::Bytes> reliable_sent;
    int resets = 0;
    bool open() override { open_ = true; return true; }
    void close() override { open_ = false; }
    bool isOpen() const override { return open_; }
    bool isDead() const override { return dead; }
    void reset(uint32_t new_conv_id) override {
        ++resets;
        conv = new_conv_id;
        open_ = true;
        last_rx = 0;
        dead = false;
        incoming.clear();
    }
    void tick(uint32_t) override {}
    int sendReliable(const core::Bytes& d) override { reliable_sent.push_back(d); return 0; }
    int sendRaw(const core::Bytes& d) override { raw_sent.push_back(d); return 0; }
    std::optional<core::Bytes> receive(uint32_t now_ms) override {
        if (incoming.empty()) return std::nullopt;
        core::Bytes b = std::move(incoming.front());
        incoming.pop_front();
        last_rx = now_ms;
        return b;
    }
    uint32_t lastRxMs() const override { return last_rx; }
    uint32_t conversationId() const override { return conv; }
};
struct FakeSource : core::IAudioSource {
    int frames_left = 3;
    std::size_t bytes_per_frame = 8;
    bool open() override { return true; }
    void close() override {}
    bool readFrame(core::Bytes& pcm) override {
        if (frames_left-- <= 0) return false;
        pcm.assign(bytes_per_frame, 0x11);
        return true;
    }
};
struct FakeEncoder : core::IAudioEncoder {
    bool init() override { return true; }
    bool encode(const core::Bytes& pcm, core::Bytes& out) override {
        out.assign(pcm.begin(), pcm.end());
        return true;
    }
    uint32_t frameDurationMs() const override { return 20; }
    uint32_t sampleRate() const override { return 16000; }
    uint32_t channelCount() const override { return 2; }
};
void test_button_state_machine() {
    using general::input::ButtonEvent;
    general::input::ButtonStateMachine b(1000);
    CHECK(b.update(false, 0) == ButtonEvent::None);
    CHECK(b.update(true, 100) == ButtonEvent::None);
    CHECK(b.update(false, 300) == ButtonEvent::ShortPress);
    CHECK(b.update(true, 1000) == ButtonEvent::None);
    CHECK(b.update(true, 2000) == ButtonEvent::LongPress);
    CHECK(b.update(true, 2100) == ButtonEvent::None);
    CHECK(b.update(false, 2200) == ButtonEvent::None);
    general::input::ButtonStateMachine c(1000);
    CHECK(c.update(true, 0) == ButtonEvent::None);
    CHECK(c.update(false, 1000) == ButtonEvent::LongPress);
}
void test_audio_pipeline_timestamps() {
    FakeSource src; FakeEncoder enc; FakeClock clk; FakeLogger log;
    std::vector<uint32_t> timestamps;
    std::vector<core::Bytes> payloads;
    core::AudioPipeline p(src, enc, clk, log,
        [&](const core::Bytes& b, uint32_t ts) {
            timestamps.push_back(ts);
            payloads.push_back(b);
        });
    CHECK(p.step());
    CHECK(p.step());
    CHECK(p.step());
    CHECK(!p.step());
    CHECK(timestamps.size() == 3);
    CHECK(!timestamps.empty() && timestamps[0] == 0);
    CHECK(timestamps.size() > 1 && timestamps[1] == 20);
    CHECK(timestamps.size() > 2 && timestamps[2] == 40);
    CHECK(!payloads.empty() && payloads[0].size() == 8);
}
void test_session_handshake_link_and_timeout() {
    FakeTransport tr;
    FakeLink link; link.up = true;
    FakeClock clk;
    FakeRandom rnd;
    FakeCrypto crypto;
    FakeLogger log;
    core::SessionManager sm(tr, link, clk, rnd, crypto, log);
    sm.start();
    sm.tick();
    CHECK(tr.raw_sent.size() == 1);
    CHECK(!sm.isEstablished());
    tr.incoming.push_back(core::Bytes{1, 2, 3, 4});
    sm.tick();
    CHECK(sm.isEstablished());
    CHECK(sm.send(core::Bytes{0x55, 0x66}) == 0);
    sm.tick();
    CHECK(tr.reliable_sent.size() == 1);
    link.up = false;
    sm.tick();
    CHECK(!sm.isEstablished());
    CHECK(sm.epoch() == 1);
    CHECK(!tr.isOpen());
    link.up = true;
    sm.tick();
    CHECK(!tr.isOpen());
    clk.advance(1100);
    sm.tick();
    CHECK(tr.isOpen());
    CHECK(tr.raw_sent.size() == 2);
    tr.incoming.push_back(core::Bytes{7});
    sm.tick();
    CHECK(sm.isEstablished());
    const uint32_t conv_before = tr.conversationId();
    clk.advance(9000);
    sm.tick();
    CHECK(!sm.isEstablished());
    CHECK(tr.resets == 1);
    CHECK(tr.conversationId() != conv_before);
    CHECK(sm.epoch() == 2);
}
